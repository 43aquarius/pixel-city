    79→const UI_KEY = 'pxc_ui_v1';
    80→
    81→function loadUi(): UiState {
    82→  try {
    83→    const raw = localStorage.getItem(UI_KEY);
    84→    if (raw) return { ...UI_DEFAULT, ...(JSON.parse(raw) as Partial<UiState>) };
    85→  } catch { /* 忽略 */ }
    86→  return { ...UI_DEFAULT };
    87→}
    88→
    89→// ---- 外部 UI 状态存储（localStorage 持久化，useSyncExternalStore 水合安全）----
    90→let uiStore: UiState = (() => {
    91→  try {
    92→    const raw = localStorage.getItem(UI_KEY);
    93→    if (raw) return { ...UI_DEFAULT, ...(JSON.parse(raw) as Partial<UiState>) };
    94→  } catch { /* SSR / 数据损坏时用默认值 */ }
    95→  return { ...UI_DEFAULT };
    96→})();
    97→const uiSubs = new Set<() => void>();
    98→
    99→function setUiStore(next: (u: UiState) => UiState): void {
   100→  uiStore = next(uiStore);
   101→  try { localStorage.setItem(UI_KEY, JSON.stringify(uiStore)); } catch { /* 忽略 */ }
   102→  for (const fn of uiSubs) fn();
   103→}
   104→
   105→function useUiState(): [UiState, typeof setUiStore] {
   106→  const subscribe = useCallback((cb: () => void) => {
   107→    uiSubs.add(cb);
   108→    return () => { uiSubs.delete(cb); };
   109→  }, []);
   110→  const snap = useSyncExternalStore(subscribe, () => uiStore, () => UI_DEFAULT);
   111→  return [snap, setUiStore];
   112→}

   114→function toolHint(t: Tool): string {
   115→  switch (t.t) {
   116→    case 'view': return '拖动平移 · 滚轮/双指缩放 · 空格居中';
   117→    case 'terr': return `按住拖动画画「${TERR_OPTS[t.k]?.label ?? '地形'}」（水域会淹没地上物）`;
   118→    case 'tree': return '点击或拖动，种下随机小树（共 6 种，含樱花与棕榈）';
   119→    case 'road': return '按住拖出路网，转角自动拼接';
   120→    case 'doze': return '点击拆除：整栋建筑 / 树木 / 道路';
   121→    case 'build': {
   122→      const d = defById(t.id);
   123→      return `点击放置「${d.name}」占地 ${d.w}×${d.h}${d.pop ? ` · 容纳 ${d.pop} 人` : ''}`;
   124→    }
   125→  }
   126→}
