    75→  constructor(beeper: Beeper) {
    76→    this.beeper = beeper;

    88→  async init(onProgress: (frac: number, label: string) => void): Promise<void> {
    89→    await this.book.init(onProgress);
    90→    this.treeSprites = [0, 1, 2, 3].map(v => renderModel(genTree(v * 977 + 13).model));
    91→    this.map.random((Math.random() * 1e9) | 0);

    95→  attach(display: HTMLCanvasElement, mini: HTMLCanvasElement): void {
    96→    this.display = display;
    97→    this.dctx = display.getContext('2d');
    98→    this.mini = mini;
    99→    mini.width = this.map.n * MM;
   100→    mini.height = this.map.n * MM;
   101→    this.bindInput(display);
   102→    this.bindMini(mini);
   103→    this.resize();
   104→    this.loop();

   107→  destroy(): void {
   108→    cancelAnimationFrame(this.raf);
   109→    for (const [t, n, f] of this.listeners) t.removeEventListener(n, f);
   110→    this.listeners = [];
   111→  }
   112→
   113→  private on(t: EventTarget, name: string, f: EventListener, opts?: AddEventListenerOptions): void {
   114→    t.addEventListener(name, f, opts);
   115→    this.listeners.push([t, name, f]);
   116→  }
   117→
   118→  resize(): void {
   119→    const d = this.display;
   120→    if (!d) return;
   121→    const rect = d.getBoundingClientRect();
   122→    this.dpr = Math.min(2, window.devicePixelRatio || 1);
   123→    const cssW = Math.max(100, rect.width);
   124→    const cssH = Math.max(100, rect.height);
   125→    d.width = Math.round(cssW * this.dpr);
   126→    d.height = Math.round(cssH * this.dpr);
   127→    this.lowres.width = Math.ceil(cssW / this.zoom);
   128→    this.lowres.height = Math.ceil(cssH / this.zoom);
   129→    this.lctx.imageSmoothingEnabled = false;
   130→    if (this.dctx) this.dctx.imageSmoothingEnabled = false;
   131→  }
   132→
   133→  // ---------------- 坐标 ----------------
   134→
   135→  private get lw(): number { return this.lowres.width; }
   136→  private get lh(): number { return this.lowres.height; }
   137→
   138→  /** 世界格角点 → 等距屏幕 px（低分辨率画布坐标） */
   139→  private w2sx(wx: number, wy: number): number { return (wx - wy) * 16 - this.cam.x + this.lw / 2; }
   140→  private w2sy(wx: number, wy: number): number { return (wx + wy) * 8 - this.cam.y + this.lh / 2; }
   141→
   142→  /** 鼠标 CSS 坐标 → 格坐标 */
   143→  tileAt(cssX: number, cssY: number): { tx: number; ty: number } {
   144→    const sx = cssX / this.zoom, sy = cssY / this.zoom;
   145→    const ax = sx + this.cam.x - this.lw / 2;   // = (X−Y)*16
   146→    const ay = sy + this.cam.y - this.lh / 2;   // = (X+Y)*8
   147→    const X = ax / 32 + ay / 16;
   148→    const Y = ay / 16 - ax / 32;
   149→    return { tx: Math.floor(X), ty: Math.floor(Y) };
   150→  }
   151→
   152→  center(): void {
   153→    const c = this.map.n / 2;
   154→    this.cam.x = (c - c) * 16;
   155→    this.cam.y = (c + c) * 8;
   156→  }

   158→  // ---------------- 工具 ----------------
   159→
   160→  setTool(t: Tool): void {
   161→    this.tool = t;
   162→    this.hooks.onTool?.(t);
   163→  }
   164→
   165→  cycleZoom(dir: number): void {
   166→    const i = ZOOMS.indexOf(this.zoom);
   167→    const j = Math.max(0, Math.min(ZOOMS.length - 1, i + dir));
   168→    if (j !== i) this.setZoom(ZOOMS[j]);
   169→  }
   170→
   171→  setZoom(z: number, focusCss?: { x: number; y: number }): void {
   172→    if (z === this.zoom) return;
   173→    let keep: { gx: number; gy: number } | null = null;
   174→    if (focusCss) {
   175→      const t = this.tileAt(focusCss.x, focusCss.y);
   176→      keep = { gx: t.tx + 0.5, gy: t.ty + 0.5 };
   177→    }
   178→    this.zoom = z;
   179→    this.resize();
   180→    if (keep && focusCss) {
   181→      const sx = focusCss.x / z, sy = focusCss.y / z;
   182→      this.cam.x = (keep.gx - keep.gy) * 16 - sx + this.lw / 2;
   183→      this.cam.y = (keep.gx + keep.gy) * 8 - sy + this.lh / 2;
   184→    }
   185→    this.beeper.play('zoom');
   186→    this.hooks.onZoom?.(z);
   187→  }
   188→
   189→  // ---------------- 主循环 ----------------
   190→
   191→  private loop = (): void => {
   192→    this.raf = requestAnimationFrame(this.loop);
   193→    const now = performance.now();
   194→    const dt = Math.min(100, now - this.lastT);
   195→    this.lastT = now;
   196→    this.fpsAcc += dt; this.fpsN++;
   197→    if (now - this.fpsAt > 500) {
   198→      this.hooks.onFps?.(Math.round(1000 / (this.fpsAcc / this.fpsN)));
   199→      this.fpsAcc = 0; this.fpsN = 0; this.fpsAt = now;
   200→    }
   201→    const pan = dt * 0.55;
   202→    if (this.keys.has('ArrowLeft') || this.keys.has('KeyA')) this.cam.x -= pan;
   203→    if (this.keys.has('ArrowRight') || this.keys.has('KeyD')) this.cam.x += pan;
   204→    if (this.keys.has('ArrowUp') || this.keys.has('KeyW')) this.cam.y -= pan;
   205→    if (this.keys.has('ArrowDown') || this.keys.has('KeyS')) this.cam.y += pan;
   206→    this.render(now);
   207→    if (this.map.version !== this.lastSaveVer && now - this.lastSaveT > 25000) {
   208→      this.autoSave();
   209→    }
   210→  };

   214→  private visibleRange(): { x0: number; y0: number; x1: number; y1: number } {
   215→    const n = this.map.n;
   216→    const cs: Array<[number, number]> = [[0, 0], [this.lw, 0], [0, this.lh], [this.lw, this.lh]];
   217→    let minX = n, maxX = 0, minY = n, maxY = 0;
   218→    for (const [sx, sy] of cs) {
   219→      const ax = sx + this.cam.x - this.lw / 2;
   220→      const ay = sy + this.cam.y - this.lh / 2;
   221→      const X = ax / 32 + ay / 16;
   222→      const Y = ay / 16 - ax / 32;
   223→      minX = Math.min(minX, Math.floor(X)); maxX = Math.max(maxX, Math.ceil(X));
   224→      minY = Math.min(minY, Math.floor(Y)); maxY = Math.max(maxY, Math.ceil(Y));
   225→    }
   226→    const pad = 4;
   227→    return {
   228→      x0: Math.max(0, minX - pad), y0: Math.max(0, minY - pad),
   229→      x1: Math.min(n - 1, maxX + pad), y1: Math.min(n - 1, maxY + pad),
   230→    };
   231→  }

   233→  private render(t: number): void {
   234→    const ctx = this.lctx;
   235→    const { lw, lh } = this;
   236→
   237→    // 背景虚空（深海蓝 + 稀疏星点抖动）
   238→    ctx.fillStyle = '#0B0F24';
   239→    ctx.fillRect(0, 0, lw, lh);
   240→    ctx.fillStyle = '#182142';
   241→    for (let y = 0; y < lh; y += 8) {
   242→      for (let x = ((y >> 3) % 2) * 8; x < lw; x += 16) ctx.fillRect(x + 3, y + 5, 1, 1);
   243→    }
   244→
   245→    const vr = this.visibleRange();
   246→    const wf = Math.floor(t / 620) % 2;
   247→    const items: RenderItem[] = [];
   248→    const m = this.map;
   249→
   250→    // ---- Pass 1：地面 ----
   251→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   252→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   253→        const i = m.idx(tx, ty);
   254→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   255→        if (sx < -96 || sx > lw + 96 || sy < -160 || sy > lh + 96) continue;
   256→
   257→        const a = m.occ[i];
   258→        let groundDrawn = false;
   259→        if (a >= 0) {
   260→          const o = m.objs.get(a)!;
   261→          const d = defById(o.id);
   262→          const isAnchor = o.ax === tx && o.ay === ty;
   263→          if (d.ground) {
   264→            if (isAnchor) {
   265→              const g = getGround(d.ground, { variant: o.v });
   266→              ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   267→            }
   268→            groundDrawn = true;   // 大地面精灵覆盖全部占地格
   269→          }
   270→        }
   271→        if (groundDrawn) continue;
   272→
   273→        if (m.road[i]) {
   274→          const g = getGround('road', { road: this.roadMask(tx, ty), variant: Math.floor(hash2(tx, ty, 5) * 4) });
   275→          ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   276→        } else {
   277→          const k = m.terr[i];
   278→          const v = Math.floor(hash2(tx, ty, 9) * 4);
   279→          if (k === TERR_WATER) {
   280→            const g = getGround('water', { edge: this.landEdge(tx, ty), variant: v, frame: wf });
   281→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   282→          } else if (k === TERR_SAND) {
   283→            const g = getGround('sand', { variant: v });
   284→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   285→          } else {
   286→            const g = getGround('grass', { variant: v, edge: this.waterEdge(tx, ty) });
   287→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   288→          }
   289→        }
   290→      }
   291→    }
   292→
   293→    // ---- Pass 2：树木 + 建筑（按前角深度排序）----
   294→    const seen = new Set<number>();
   295→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   296→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   297→        const i = m.idx(tx, ty);
   298→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   299→        if (sx < -160 || sx > lw + 160 || sy < -300 || sy > lh + 160) continue;
   300→
   301→        const tv = m.tree[i];
   302→        if (tv >= 0 && !m.road[i] && m.occ[i] < 0) {
   303→          const sp = this.treeSprites[tv & 3];
   304→          const key = tx + ty;
   305→          const dx = Math.round(sx + sp.ox), dy = Math.round(sy + sp.oy);
   306→          items.push({ key, draw: () => ctx.drawImage(sp.cv, dx, dy) });
   307→        }
   308→
   309→        const a = m.occ[i];
   310→        if (a >= 0 && !seen.has(a)) {
   311→          seen.add(a);
   312→          const o = m.objs.get(a)!;
   313→          const d = defById(o.id);
   314→          const bs = this.book.get(o.id, o.v);
   315→          const sp = bs.sp;
   316→          const bsx = this.w2sx(o.ax, o.ay), bsy = this.w2sy(o.ax, o.ay);
   317→          const key = (o.ax + d.w - 1) + (o.ay + d.h - 1);
   318→          const blinkOn = Math.floor(t / 640) % 2 === 0;
   319→          const smokeFr = Math.floor(t / 300) % 3;
   320→          const bx = Math.round(bsx + sp.ox), by = Math.round(bsy + sp.oy);
   321→          const smk = bs.smokeAt;
   322→          const blk = bs.blinkAt;
   323→          const sm = smk ? this.book.smoke[smokeFr] : null;
   324→          const mx = smk ? bsx + (smk[0] - smk[1]) : 0;
   325→          const my = smk ? bsy + (smk[0] + smk[1]) / 2 - smk[2] : 0;
   326→          const lx = blk ? bsx + (blk[0] - blk[1]) : 0;
   327→          const ly = blk ? bsy + (blk[0] + blk[1]) / 2 - blk[2] : 0;
   328→          items.push({
   329→            key,
   330→            draw: () => {
   331→              ctx.drawImage(sp.cv, bx, by);
   332→              if (sm) ctx.drawImage(sm.cv, Math.round(mx + sm.ox), Math.round(my + sm.oy));
   333→              if (blk && blinkOn) {
   334→                ctx.fillStyle = PALETTE[62];
   335→                ctx.fillRect(Math.round(lx) - 1, Math.round(ly) - 1, 2, 2);
   336→                ctx.fillStyle = PALETTE[61];
   337→                ctx.fillRect(Math.round(lx), Math.round(ly), 1, 1);
   338→              }
   339→            },
   340→          });
   341→        }
   342→      }
   343→    }
   344→    items.sort((p, q) => p.key - q.key);
   345→    for (const it of items) it.draw();
   346→
   347→    // ---- Pass 3：光标 + 幽灵 ----
   348→    this.drawCursor(ctx);
   349→
   350→    // ---- 上屏（最近邻放大 + 扫描线）----
   351→    const d = this.display;
   352→    const dctx = this.dctx;
   353→    if (d && dctx) {
   354→      const S = this.zoom * this.dpr;
   355→      dctx.imageSmoothingEnabled = false;
   356→      dctx.drawImage(this.lowres, 0, 0, lw, lh, 0, 0, Math.round(lw * S), Math.round(lh * S));
   357→      const step = Math.max(2, Math.round(S));
   358→      dctx.globalAlpha = 0.15;
   359→      dctx.fillStyle = '#000';
   360→      for (let y = step - 1; y < d.height; y += step) dctx.fillRect(0, y, d.width, 1);
   361→      dctx.globalAlpha = 1;
   362→    }
   363→
   364→    this.renderMini();
   365→  }
   366→
   367→  private roadMask(tx: number, ty: number): number {
   368→    const m = this.map;
   369→    let v = 0;
   370→    if (m.inB(tx, ty - 1) && m.road[m.idx(tx, ty - 1)]) v |= 1;
   371→    if (m.inB(tx + 1, ty) && m.road[m.idx(tx + 1, ty)]) v |= 2;
   372→    if (m.inB(tx, ty + 1) && m.road[m.idx(tx, ty + 1)]) v |= 4;
   373→    if (m.inB(tx - 1, ty) && m.road[m.idx(tx - 1, ty)]) v |= 8;
   374→    return v;
   375→  }
   376→
   377→  private waterEdge(tx: number, ty: number): number {
   378→    const m = this.map;
   379→    let v = 0;
   380→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] === TERR_WATER) v |= 1;
   381→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] === TERR_WATER) v |= 2;
   382→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] === TERR_WATER) v |= 4;
   383→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] === TERR_WATER) v |= 8;
   384→    return v;
   385→  }
   386→
   387→  private landEdge(tx: number, ty: number): number {
   388→    const m = this.map;
   389→    let v = 0;
   390→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] !== TERR_WATER) v |= 1;
   391→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] !== TERR_WATER) v |= 2;
   392→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] !== TERR_WATER) v |= 4;
   393→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] !== TERR_WATER) v |= 8;
   394→    return v;
   395→  }

   397→  /** 屏幕偏移是否在 w×h 足印菱形内（相对锚点） */
   398→  private inFootprint(dx: number, dy: number, w: number, h: number): boolean {
   399→    const X = dx / 2 + dy;
   400→    const Y = dy - dx / 2;
   401→    return X >= 0 && X <= 16 * w && Y >= 0 && Y <= 16 * h;
   402→  }
   403→
   404→  /** 光标菱形 + 放置幽灵 */
   405→  private drawCursor(ctx: CanvasRenderingContext2D): void {
   406→    const h = this.hover;
   407→    if (!h || !this.map.inB(h.tx, h.ty)) return;
   408→    const tool = this.tool;
   409→    let w = 1, hh = 1;
   410→    if (tool.t === 'build') {
   411→      const d = defById(tool.id);
   412→      w = d.w; hh = d.h;
   413→    }
   414→    const x0 = this.w2sx(h.tx, h.ty), y0 = this.w2sy(h.tx, h.ty);
   415→    ctx.beginPath();
   416→    ctx.moveTo(x0, y0);
   417→    ctx.lineTo(x0 + 16 * w, y0 + 8 * w);
   418→    ctx.lineTo(x0 + 16 * (w - hh), y0 + 8 * (w + hh));
   419→    ctx.lineTo(x0 - 16 * hh, y0 + 8 * hh);
   420→    ctx.closePath();
   421→    ctx.strokeStyle = 'rgba(8,10,18,0.85)';
   422→    ctx.lineWidth = 3;
   423→    ctx.stroke();
   424→    ctx.strokeStyle = '#F4F2E8';
   425→    ctx.lineWidth = 1;
   426→    ctx.stroke();
   427→
   428→    if (tool.t === 'build') {
   429→      const bs = this.book.get(tool.id, tool.variant);
   430→      const g = ghostSprite(bs.sp);
   431→      const ok = this.map.canPlace(tool.id, h.tx, h.ty);
   432→      ctx.drawImage(g, Math.round(x0 + bs.sp.ox), Math.round(y0 + bs.sp.oy));
   433→      if (!ok) {
   434→        ctx.fillStyle = 'rgba(224,85,72,0.55)';
   435→        const W = 16 * (w + hh), H = 8 * (w + hh) + 2;
   436→        for (let yy = 0; yy < H; yy += 2) {
   437→          for (let xx = ((yy >> 1) % 2) * 2; xx < W * 2; xx += 4) {
   438→            const px = x0 - 16 * hh + xx, py = y0 + yy;
   439→            if (this.inFootprint(px - x0, py - y0, w, hh)) ctx.fillRect(px, py, 2, 1);
   440→          }
   441→        }
   442→      }
   443→    }
   444→  }
   445→
   446→  // ---------------- 小地图 ----------------
   447→
   448→  private renderMini(): void {
   449→    const mini = this.mini;
   450→    if (!mini) return;
   451→    const mctx = mini.getContext('2d')!;
   452→    const n = this.map.n;
   453→    if (this.miniDirty) {
   454→      this.miniBase.width = n * MM;
   455→      this.miniBase.height = n * MM;
   456→      const bctx = this.miniBase.getContext('2d')!;
   457→      for (let y = 0; y < n; y++) {
   458→        for (let x = 0; x < n; x++) {
   459→          const i = this.map.idx(x, y);
   460→          let c = '#3B7138';
   461→          const k = this.map.terr[i];
   462→          if (k === TERR_WATER) c = '#3467AC';
   463→          else if (k === TERR_SAND) c = '#CFB379';
   464→          if (this.map.tree[i] >= 0) c = '#245129';
   465→          if (this.map.road[i]) c = '#4A4C55';
   466→          const a = this.map.occ[i];
   467→          if (a >= 0) {
   468→            const o = this.map.objs.get(a)!;
   469→            const cat = defById(o.id).cat;
   470→            c = cat === 'res' ? '#C25B44' : cat === 'com' ? '#E8CC4C'
   471→              : cat === 'ind' ? '#85878F' : '#5BC8B8';
   472→          }
   473→          bctx.fillStyle = c;
   474→          bctx.fillRect(x * MM, y * MM, MM, MM);
   475→        }
   476→      }
   477→      this.miniDirty = false;
   478→    }
   479→    mctx.imageSmoothingEnabled = false;
   480→    mctx.drawImage(this.miniBase, 0, 0);
   481→    const vr = this.visibleRange();
   482→    mctx.strokeStyle = '#F4F2E8';
   483→    mctx.lineWidth = 1;
   484→    mctx.strokeRect(vr.x0 * MM + 0.5, vr.y0 * MM + 0.5, (vr.x1 - vr.x0) * MM, (vr.y1 - vr.y0) * MM);
   485→  }

   487→  private bindMini(mini: HTMLCanvasElement): void {
   488→    const jump = (e: PointerEvent): void => {
   489→      const rect = mini.getBoundingClientRect();
   490→      const tx = Math.floor(((e.clientX - rect.left) / rect.width) * this.map.n);
   491→      const ty = Math.floor(((e.clientY - rect.top) / rect.height) * this.map.n);
   492→      this.cam.x = (tx - ty) * 16;
   493→      this.cam.y = (tx + ty) * 8;
   494→    };
   495→    this.on(mini, 'pointerdown', (e) => {
   496→      jump(e as PointerEvent);
   497→      const mv = (ev: Event) => jump(ev as PointerEvent);
   498→      const up = () => {
   499→        mini.removeEventListener('pointermove', mv);
   500→        mini.removeEventListener('pointerup', up);
   501→      };
   502→      mini.addEventListener('pointermove', mv);
   503→      mini.addEventListener('pointerup', up);
   504→    });
   505→  }

   509→  private bindInput(display: HTMLCanvasElement): void {
   510→    this.on(display, 'contextmenu', (e) => e.preventDefault());
   511→    this.on(display, 'pointerdown', (e) => {
   512→      const pe = e as PointerEvent;
   513→      display.setPointerCapture(pe.pointerId);
   514→      const rect = display.getBoundingClientRect();
   515→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   516→      this.beeper.play('click');
   517→      if (pe.button === 1 || pe.button === 2 || this.tool.t === 'view') {
   518→        this.panning = true;
   519→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   520→        return;
   521→      }
   522→      if (pe.button !== 0) return;
   523→      const { tx, ty } = this.tileAt(cx, cy);
   524→      this.pushUndo();
   525→      this.painting = true;
   526→      this.lastTile = { x: tx, y: ty };
   527→      this.applyToolAt(tx, ty);
   528→    });
   529→    this.on(display, 'pointermove', (e) => {
   530→      const pe = e as PointerEvent;
   531→      const rect = display.getBoundingClientRect();
   532→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   533→      const { tx, ty } = this.tileAt(cx, cy);
   534→      this.hover = { tx, ty };
   535→      this.hooks.onHover?.(tx, ty);
   536→      if (this.panning) {
   537→        this.cam.x -= (pe.clientX - this.lastPan.x) / this.zoom;
   538→        this.cam.y -= (pe.clientY - this.lastPan.y) / this.zoom;
   539→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   540→        return;
   541→      }
   542→      if (this.painting) {
   543→        const lt = this.lastTile;
   544→        if (!lt) {
   545→          this.lastTile = { x: tx, y: ty };
   546→          this.dragMode = true;
   547→          this.applyToolAt(tx, ty);
   548→          this.dragMode = false;
   549→          return;
   550→        }
   551→        const steps = Math.max(Math.abs(tx - lt.x), Math.abs(ty - lt.y));
   552→        if (steps > 0) {
   553→          this.dragMode = true;
   554→          for (let s = 1; s <= steps; s++) {
   555→            const ix = Math.round(lt.x + (tx - lt.x) * s / steps);
   556→            const iy = Math.round(lt.y + (ty - lt.y) * s / steps);
   557→            this.applyToolAt(ix, iy);
   558→          }
   559→          this.dragMode = false;
   560→          this.lastTile = { x: tx, y: ty };
   561→        }
   562→      }
   563→    });
   564→    const endPointer = (): void => {
   565→      this.painting = false;
   566→      this.panning = false;
   567→      this.lastTile = null;
   568→    };
   569→    this.on(display, 'pointerup', endPointer);
   570→    this.on(display, 'pointercancel', endPointer);
   571→    this.on(display, 'pointerleave', () => { this.hover = null; });
   572→    this.on(display, 'wheel', (e) => {
   573→      const we = e as WheelEvent;
   574→      we.preventDefault();
   575→      const rect = display.getBoundingClientRect();
   576→      this.setZoom(
   577→        ZOOMS[Math.max(0, Math.min(ZOOMS.length - 1, ZOOMS.indexOf(this.zoom) + (we.deltaY < 0 ? 1 : -1)))],
   578→        { x: we.clientX - rect.left, y: we.clientY - rect.top },
   579→      );
   580→    }, { passive: false });
   581→
   582→    this.on(window, 'keydown', (e) => {
   583→      const ke = e as KeyboardEvent;
   584→      if (ke.target instanceof HTMLInputElement || ke.target instanceof HTMLTextAreaElement) return;
   585→      this.keys.add(ke.code);
   586→      if (ke.code === 'Space') { e.preventDefault(); this.center(); }
   587→      if (ke.code === 'Equal' || ke.code === 'NumpadAdd') this.cycleZoom(1);
   588→      if (ke.code === 'Minus' || ke.code === 'NumpadSubtract') this.cycleZoom(-1);
   589→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyZ') {
   590→        e.preventDefault();
   591→        if (ke.shiftKey) this.redo(); else this.undo();
   592→      }
   593→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyY') {
   594→        e.preventDefault();
   595→        this.redo();
   596→      }
   597→      if (ke.code === 'Escape') this.setTool({ t: 'view' });
   598→    });
   599→    this.on(window, 'keyup', (e) => this.keys.delete((e as KeyboardEvent).code));
   600→  }

   604→  private applyToolAt(tx: number, ty: number): void {
   605→    const t = this.tool;
   606→    switch (t.t) {
   607→      case 'terr':
   608→        if (this.map.setTerr(tx, ty, t.k)) this.beeper.play('paint');
   609→        break;
   610→      case 'road':
   611→        if (this.map.setRoad(tx, ty, true)) this.beeper.play('paint');
   612→        break;
   613→      case 'tree':
   614→        if (this.map.plantTree(tx, ty)) this.beeper.play('paint');
   615→        break;
   616→      case 'doze':
   617→        if (this.map.bulldoze(tx, ty)) this.beeper.play('delete');
   618→        break;
   619→      case 'build': {
   620→        if (this.map.place(t.id, t.variant, tx, ty)) {
   621→          this.beeper.play('place');
   622→        } else if (!this.dragMode && this.map.inB(tx, ty)) {
   623→          this.beeper.play('error');
   624→        }
   625→        break;
   626→      }
   627→    }
   628→  }

   632→  pushUndo(): void {
   633→    this.undoStack.push(this.map.serialize());
   634→    if (this.undoStack.length > 24) this.undoStack.shift();
   635→    this.redoStack.length = 0;
   636→  }
   637→
   638→  canUndo(): boolean { return this.undoStack.length > 0; }
   639→  canRedo(): boolean { return this.redoStack.length > 0; }
   640→
   641→  undo(): void {
   642→    const s = this.undoStack.pop();
   643→    if (!s) { this.beeper.play('error'); return; }
   644→    this.redoStack.push(this.map.serialize());
   645→    this.map.deserialize(s);
   646→    this.beeper.play('click');
   647→  }
   648→
   649→  redo(): void {
   650→    const s = this.redoStack.pop();
   651→    if (!s) { this.beeper.play('error'); return; }
   652→    this.undoStack.push(this.map.serialize());
   653→    this.map.deserialize(s);
   654→    this.beeper.play('click');
   655→  }

   670→  saveSlot(i: number): boolean {
   671→    try {
   672→      localStorage.setItem(Engine.SLOT_KEY[i], this.map.serialize());
   673→      const st = this.map.stats();
   674→      localStorage.setItem(`pxc_meta${i + 1}`, JSON.stringify({
   675→        time: new Date().toLocaleString('zh-CN'),
   676→        pop: st.pop,
   677→      }));
   678→      this.lastSaveVer = this.map.version;
   679→      this.beeper.play('save');
   680→      return true;
   681→    } catch {
   682→      return false;
   683→    }
   684→  }

   686→  loadSlot(i: number): boolean {
   687→    const s = localStorage.getItem(Engine.SLOT_KEY[i]);
   688→    if (!s) { this.beeper.play('error'); return false; }
   689→    if (this.map.deserialize(s)) {
   690→      this.beeper.play('open');
   691→      return true;
   692→    }
   693→    this.beeper.play('error');
   694→    return false;
   695→  }

   704→    }
   705→  }
