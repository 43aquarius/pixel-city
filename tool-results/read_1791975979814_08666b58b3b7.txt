   138→   138→  /** 世界格角点 → 等距屏幕 px（低分辨率画布坐标） */
   139→   139→  private w2sx(wx: number, wy: number): number { return (wx - wy) * 16 - this.cam.x + this.lw / 2; }

   143→   143→  tileAt(cssX: number, cssY: number): { tx: number; ty: number } {
   144→   144→    const sx = cssX / this.zoom, sy = cssY / this.zoom;
   145→   145→    const ax = sx + this.cam.x - this.lw / 2;   // = (X−Y)*16
   146→   146→    const ay = sy + this.cam.y - this.lh / 2;   // = (X+Y)*8
   147→   147→    const X = ax / 32 + ay / 16;
   148→   148→    const Y = ay / 16 - ax / 32;
   149→   149→    return { tx: Math.floor(X), ty: Math.floor(Y) };

   171→   171→  setZoom(z: number, focusCss?: { x: number; y: number }): void {
   172→   172→    if (z === this.zoom) return;
   173→   173→    let keep: { gx: number; gy: number } | null = null;
   174→   174→    if (focusCss) {
   175→   175→      const t = this.tileAt(focusCss.x, focusCss.y);
   176→   176→      keep = { gx: t.tx + 0.5, gy: t.ty + 0.5 };
   177→   177→    }
   178→   178→    this.zoom = z;
   179→   179→    this.resize();
   180→   180→    if (keep && focusCss) {
   181→   181→      const sx = focusCss.x / z, sy = focusCss.y / z;
   182→   182→      this.cam.x = (keep.gx - keep.gy) * 16 - sx + this.lw / 2;
   183→   183→      this.cam.y = (keep.gx + keep.gy) * 8 - sy + this.lh / 2;
   184→   184→    }
   185→   185→    this.beeper.play('zoom');
   186→   186→    this.hooks.onZoom?.(z);
   187→   187→  }

   191→   191→  private loop = (): void => {
   192→   192→    this.raf = requestAnimationFrame(this.loop);
   193→   193→    const now = performance.now();
   194→   194→    const dt = Math.min(100, now - this.lastT);
   195→   195→    this.lastT = now;
   196→   196→    this.fpsAcc += dt; this.fpsN++;
   197→   197→    if (now - this.fpsAt > 500) {
   198→   198→      this.hooks.onFps?.(Math.round(1000 / (this.fpsAcc / this.fpsN)));
   199→   199→      this.fpsAcc = 0; this.fpsN = 0; this.fpsAt = now;
   200→   200→    }
   201→   201→    const pan = dt * 0.55;
   202→   202→    if (this.keys.has('ArrowLeft') || this.keys.has('KeyA')) this.cam.x -= pan;
   203→   203→    if (this.keys.has('ArrowRight') || this.keys.has('KeyD')) this.cam.x += pan;
   204→   204→    if (this.keys.has('ArrowUp') || this.keys.has('KeyW')) this.cam.y -= pan;
   205→   205→    if (this.keys.has('ArrowDown') || this.keys.has('KeyS')) this.cam.y += pan;
   206→   206→    this.render(now);
   207→   207→    if (this.map.version !== this.lastSaveVer && now - this.lastSaveT > 25000) {
   208→   208→      this.autoSave();
   209→   209→    }
   210→   210→  };

   214→   214→  private visibleRange(): { x0: number; y0: number; x1: number; y1: number } {
   215→   215→    const n = this.map.n;
   216→   216→    const cs: Array<[number, number]> = [[0, 0], [this.lw, 0], [0, this.lh], [this.lw, this.lh]];
   217→   217→    let minX = n, maxX = 0, minY = n, maxY = 0;
   218→   218→    for (const [sx, sy] of cs) {
   219→   219→      const ax = sx + this.cam.x - this.lw / 2;
   220→   220→      const ay = sy + this.cam.y - this.lh / 2;
   221→   221→      const X = ax / 32 + ay / 16;
   222→   222→      const Y = ay / 16 - ax / 32;
   223→   223→      minX = Math.min(minX, Math.floor(X)); maxX = Math.max(maxX, Math.ceil(X));
   224→   224→      minY = Math.min(minY, Math.floor(Y)); maxY = Math.max(maxY, Math.ceil(Y));
   225→   225→    }
   226→   226→    const pad = 4;
   227→   227→    return {
   228→   228→      x0: Math.max(0, minX - pad), y0: Math.max(0, minY - pad),
   229→   229→      x1: Math.min(n - 1, maxX + pad), y1: Math.min(n - 1, maxY + pad),
   230→   230→    };
   231→   231→  }

   233→   233→  private render(t: number): void {
   234→   234→    const ctx = this.lctx;
   235→   235→    const { lw, lh } = this;
   236→   236→
   237→   237→    // 背景虚空（深海蓝 + 稀疏星点抖动）
   238→   238→    ctx.fillStyle = '#0B0F24';
   239→   239→    ctx.fillRect(0, 0, lw, lh);
   240→   240→    ctx.fillStyle = '#182142';
   241→   241→    for (let y = 0; y < lh; y += 8) {
   242→   242→      for (let x = ((y >> 3) % 2) * 8; x < lw; x += 16) ctx.fillRect(x + 3, y + 5, 1, 1);
   243→   243→    }
   244→   244→
   245→   245→    const vr = this.visibleRange();
   246→   246→    const wf = Math.floor(t / 620) % 2;
   247→   247→    const items: RenderItem[] = [];
   248→   248→    const m = this.map;
   249→   249→
   250→   250→    // ---- Pass 1：地面 ----
   251→   251→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   252→   252→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   253→   253→        const i = m.idx(tx, ty);
   254→   254→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   255→   255→        if (sx < -96 || sx > lw + 96 || sy < -160 || sy > lh + 96) continue;
   256→   256→
   257→   257→        const a = m.occ[i];
   258→   258→        let groundDrawn = false;
   259→   259→        if (a >= 0) {
   260→   260→          const o = m.objs.get(a)!;
   261→   261→          const d = defById(o.id);
   262→   262→          const isAnchor = o.ax === tx && o.ay === ty;
   263→   263→          if (d.ground) {
   264→   264→            if (isAnchor) {
   265→   265→              const g = getGround(d.ground, { variant: o.v });
   266→   266→              ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   267→   267→            }
   268→   268→            groundDrawn = true;   // 大地面精灵覆盖全部占地格
   269→   269→          }
   270→   270→        }
   271→   271→        if (groundDrawn) continue;
   272→   272→
   273→   273→        if (m.road[i]) {
   274→   274→          const g = getGround('road', { road: this.roadMask(tx, ty), variant: Math.floor(hash2(tx, ty, 5) * 4) });
   275→   275→          ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   276→   276→        } else {
   277→   277→          const k = m.terr[i];
   278→   278→          const v = Math.floor(hash2(tx, ty, 9) * 4);
   279→   279→          if (k === TERR_WATER) {
   280→   280→            const g = getGround('water', { edge: this.landEdge(tx, ty), variant: v, frame: wf });
   281→   281→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   282→   282→          } else if (k === TERR_SAND) {
   283→   283→            const g = getGround('sand', { variant: v });
   284→   284→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   285→   285→          } else {
   286→   286→            const g = getGround('grass', { variant: v, edge: this.waterEdge(tx, ty) });
   287→   287→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   288→   288→          }
   289→   289→        }
   290→   290→      }
   291→   291→    }
   292→   292→
   293→   293→    // ---- Pass 2：树木 + 建筑（按前角深度排序）----
   294→   294→    const seen = new Set<number>();
   295→   295→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   296→   296→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   297→   297→        const i = m.idx(tx, ty);
   298→   298→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   299→   299→        if (sx < -160 || sx > lw + 160 || sy < -300 || sy > lh + 160) continue;
   300→   300→
   301→   301→        const tv = m.tree[i];
   302→   302→        if (tv >= 0 && !m.road[i] && m.occ[i] < 0) {
   303→   303→          const sp = this.treeSprites[tv & 3];
   304→   304→          const key = tx + ty;
   305→   305→          const dx = Math.round(sx + sp.ox), dy = Math.round(sy + sp.oy);
   306→   306→          items.push({ key, draw: () => ctx.drawImage(sp.cv, dx, dy) });
   307→   307→        }
   308→   308→
   309→   309→        const a = m.occ[i];
   310→   310→        if (a >= 0 && !seen.has(a)) {
   311→   311→          seen.add(a);
   312→   312→          const o = m.objs.get(a)!;
   313→   313→          const d = defById(o.id);
   314→   314→          const bs = this.book.get(o.id, o.v);
   315→   315→          const sp = bs.sp;
   316→   316→          const bsx = this.w2sx(o.ax, o.ay), bsy = this.w2sy(o.ax, o.ay);
   317→   317→          const key = (o.ax + d.w - 1) + (o.ay + d.h - 1);
   318→   318→          const blinkOn = Math.floor(t / 640) % 2 === 0;
   319→   319→          const smokeFr = Math.floor(t / 300) % 3;
   320→   320→          const bx = Math.round(bsx + sp.ox), by = Math.round(bsy + sp.oy);
   321→   321→          const smk = bs.smokeAt;
   322→   322→          const blk = bs.blinkAt;
   323→   323→          const sm = smk ? this.book.smoke[smokeFr] : null;
   324→   324→          const mx = smk ? bsx + (smk[0] - smk[1]) : 0;
   325→   325→          const my = smk ? bsy + (smk[0] + smk[1]) / 2 - smk[2] : 0;
   326→   326→          const lx = blk ? bsx + (blk[0] - blk[1]) : 0;
   327→   327→          const ly = blk ? bsy + (blk[0] + blk[1]) / 2 - blk[2] : 0;
   328→   328→          items.push({
   329→   329→            key,
   330→   330→            draw: () => {
   331→   331→              ctx.drawImage(sp.cv, bx, by);
   332→   332→              if (sm) ctx.drawImage(sm.cv, Math.round(mx + sm.ox), Math.round(my + sm.oy));
   333→   333→              if (blk && blinkOn) {
   334→   334→                ctx.fillStyle = PALETTE[62];
   335→   335→                ctx.fillRect(Math.round(lx) - 1, Math.round(ly) - 1, 2, 2);
   336→   336→                ctx.fillStyle = PALETTE[61];
   337→   337→                ctx.fillRect(Math.round(lx), Math.round(ly), 1, 1);
   338→   338→              }
   339→   339→            },
   340→   340→          });
   341→   341→        }
   342→   342→      }
   343→   343→    }
   344→   344→    items.sort((p, q) => p.key - q.key);
   345→   345→    for (const it of items) it.draw();
   346→   346→
   347→   347→    // ---- Pass 3：光标 + 幽灵 ----
   348→   348→    this.drawCursor(ctx);
   349→   349→
   350→   350→    // ---- 上屏（最近邻放大 + 扫描线）----
   351→   351→    const d = this.display;
   352→   352→    const dctx = this.dctx;
   353→   353→    if (d && dctx) {
   354→   354→      const S = this.zoom * this.dpr;
   355→   355→      dctx.imageSmoothingEnabled = false;
   356→   356→      dctx.drawImage(this.lowres, 0, 0, lw, lh, 0, 0, Math.round(lw * S), Math.round(lh * S));
   357→   357→      const step = Math.max(2, Math.round(S));
   358→   358→      dctx.globalAlpha = 0.15;
   359→   359→      dctx.fillStyle = '#000';
   360→   360→      for (let y = step - 1; y < d.height; y += step) dctx.fillRect(0, y, d.width, 1);
   361→   361→      dctx.globalAlpha = 1;
   362→   362→    }
   363→   363→
   364→   364→    this.renderMini();
   365→   365→  }
   366→   366→
   367→   367→  private roadMask(tx: number, ty: number): number {
   368→   368→    const m = this.map;
   369→   369→    let v = 0;
   370→   370→    if (m.inB(tx, ty - 1) && m.road[m.idx(tx, ty - 1)]) v |= 1;
   371→   371→    if (m.inB(tx + 1, ty) && m.road[m.idx(tx + 1, ty)]) v |= 2;
   372→   372→    if (m.inB(tx, ty + 1) && m.road[m.idx(tx, ty + 1)]) v |= 4;
   373→   373→    if (m.inB(tx - 1, ty) && m.road[m.idx(tx - 1, ty)]) v |= 8;
   374→   374→    return v;
   375→   375→  }
   376→   376→
   377→   377→  private waterEdge(tx: number, ty: number): number {
   378→   378→    const m = this.map;
   379→   379→    let v = 0;
   380→   380→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] === TERR_WATER) v |= 1;
   381→   381→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] === TERR_WATER) v |= 2;
   382→   382→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] === TERR_WATER) v |= 4;
   383→   383→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] === TERR_WATER) v |= 8;
   384→   384→    return v;
   385→   385→  }
   386→   386→
   387→   387→  private landEdge(tx: number, ty: number): number {
   388→   388→    const m = this.map;
   389→   389→    let v = 0;
   390→   390→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] !== TERR_WATER) v |= 1;
   391→   391→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] !== TERR_WATER) v |= 2;
   392→   392→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] !== TERR_WATER) v |= 4;
   393→   393→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] !== TERR_WATER) v |= 8;
   394→   394→    return v;
   395→   395→  }

   397→   397→  /** 屏幕偏移是否在 w×h 足印菱形内（相对锚点） */
   398→   398→  private inFootprint(dx: number, dy: number, w: number, h: number): boolean {
   399→   399→    const X = dx / 2 + dy;
   400→   400→    const Y = dy - dx / 2;
   401→   401→    return X >= 0 && X <= 16 * w && Y >= 0 && Y <= 16 * h;
   402→   402→  }

   404→   404→  /** 光标菱形 + 放置幽灵 */
   405→   405→  private drawCursor(ctx: CanvasRenderingContext2D): void {
   406→   406→    const h = this.hover;
   407→   407→    if (!h || !this.map.inB(h.tx, h.ty)) return;
   408→   408→    const tool = this.tool;
   409→   409→    let w = 1, hh = 1;
   410→   410→    if (tool.t === 'build') {
   411→   411→      const d = defById(tool.id);
   412→   412→      w = d.w; hh = d.h;
   413→   413→    }
   414→   414→    const x0 = this.w2sx(h.tx, h.ty), y0 = this.w2sy(h.tx, h.ty);
   415→   415→    ctx.beginPath();
   416→   416→    ctx.moveTo(x0, y0);
   417→   417→    ctx.lineTo(x0 + 16 * w, y0 + 8 * w);
   418→   418→    ctx.lineTo(x0 + 16 * (w - hh), y0 + 8 * (w + hh));
   419→   419→    ctx.lineTo(x0 - 16 * hh, y0 + 8 * hh);
   420→   420→    ctx.closePath();
   421→   421→    ctx.strokeStyle = 'rgba(8,10,18,0.85)';
   422→   422→    ctx.lineWidth = 3;
   423→   423→    ctx.stroke();
   424→   424→    ctx.strokeStyle = '#F4F2E8';
   425→   425→    ctx.lineWidth = 1;
   426→   426→    ctx.stroke();
   427→   427→
   428→   428→    if (tool.t === 'build') {
   429→   429→      const bs = this.book.get(tool.id, tool.variant);
   430→   430→      const g = ghostSprite(bs.sp);
   431→   431→      const ok = this.map.canPlace(tool.id, h.tx, h.ty);
   432→   432→      ctx.drawImage(g, Math.round(x0 + bs.sp.ox), Math.round(y0 + bs.sp.oy));
   433→   433→      if (!ok) {
   434→   434→        ctx.fillStyle = 'rgba(224,85,72,0.55)';
   435→   435→        const W = 16 * (w + hh), H = 8 * (w + hh) + 2;
   436→   436→        for (let yy = 0; yy < H; yy += 2) {
   437→   437→          for (let xx = ((yy >> 1) % 2) * 2; xx < W * 2; xx += 4) {
   438→   438→            const px = x0 - 16 * hh + xx, py = y0 + yy;
   439→   439→            if (this.inFootprint(px - x0, py - y0, w, hh)) ctx.fillRect(px, py, 2, 1);
   440→   440→          }
   441→   441→        }
   442→   442→      }
   443→   443→    }
   444→   444→  }

   448→   448→  private renderMini(): void {
   449→   449→    const mini = this.mini;
   450→   450→    if (!mini) return;
   451→   451→    const mctx = mini.getContext('2d')!;
   452→   452→    const n = this.map.n;
   453→   453→    if (this.miniDirty) {
   454→   454→      this.miniBase.width = n * MM;
   455→   455→      this.miniBase.height = n * MM;
   456→   456→      const bctx = this.miniBase.getContext('2d')!;
   457→   457→      for (let y = 0; y < n; y++) {
   458→   458→        for (let x = 0; x < n; x++) {
   459→   459→          const i = this.map.idx(x, y);
   460→   460→          let c = '#3B7138';
   461→   461→          const k = this.map.terr[i];
   462→   462→          if (k === TERR_WATER) c = '#3467AC';
   463→   463→          else if (k === TERR_SAND) c = '#CFB379';
   464→   464→          if (this.map.tree[i] >= 0) c = '#245129';
   465→   465→          if (this.map.road[i]) c = '#4A4C55';
   466→   466→          const a = this.map.occ[i];
   467→   467→          if (a >= 0) {
   468→   468→            const o = this.map.objs.get(a)!;
   469→   469→            const cat = defById(o.id).cat;
   470→   470→            c = cat === 'res' ? '#C25B44' : cat === 'com' ? '#E8CC4C'
   471→   471→              : cat === 'ind' ? '#85878F' : '#5BC8B8';
   472→   472→          }
   473→   473→          bctx.fillStyle = c;
   474→   474→          bctx.fillRect(x * MM, y * MM, MM, MM);
   475→   475→        }
   476→   476→      }
   477→   477→      this.miniDirty = false;
   478→   478→    }
   479→   479→    mctx.imageSmoothingEnabled = false;
   480→   480→    mctx.drawImage(this.miniBase, 0, 0);
   481→   481→    const vr = this.visibleRange();
   482→   482→    mctx.strokeStyle = '#F4F2E8';
   483→   483→    mctx.lineWidth = 1;
   484→   484→    mctx.strokeRect(vr.x0 * MM + 0.5, vr.y0 * MM + 0.5, (vr.x1 - vr.x0) * MM, (vr.y1 - vr.y0) * MM);
   485→   485→  }

   487→   487→  private bindMini(mini: HTMLCanvasElement): void {
   488→   488→    const jump = (e: PointerEvent): void => {
   489→   489→      const rect = mini.getBoundingClientRect();
   490→   490→      const tx = Math.floor(((e.clientX - rect.left) / rect.width) * this.map.n);
   491→   491→      const ty = Math.floor(((e.clientY - rect.top) / rect.height) * this.map.n);
   492→   492→      this.cam.x = (tx - ty) * 16;
   493→   493→      this.cam.y = (tx + ty) * 8;
   494→   494→    };
   495→   495→    this.on(mini, 'pointerdown', (e) => {
   496→   496→      jump(e as PointerEvent);
   497→   497→      const mv = (ev: Event) => jump(ev as PointerEvent);
   498→   498→      const up = () => {
   499→   499→        mini.removeEventListener('pointermove', mv);
   500→   500→        mini.removeEventListener('pointerup', up);
   501→   501→      };
   502→   502→      mini.addEventListener('pointermove', mv);
   503→   503→      mini.addEventListener('pointerup', up);
   504→   504→    });
   505→   505→  }

   509→   509→  private bindInput(display: HTMLCanvasElement): void {
   510→   510→    this.on(display, 'contextmenu', (e) => e.preventDefault());
   511→   511→    this.on(display, 'pointerdown', (e) => {
   512→   512→      const pe = e as PointerEvent;
   513→   513→      display.setPointerCapture(pe.pointerId);
   514→   514→      const rect = display.getBoundingClientRect();
   515→   515→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   516→   516→      this.beeper.play('click');
   517→   517→      if (pe.button === 1 || pe.button === 2 || this.tool.t === 'view') {
   518→   518→        this.panning = true;
   519→   519→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   520→   520→        return;
   521→   521→      }
   522→   522→      if (pe.button !== 0) return;
   523→   523→      const { tx, ty } = this.tileAt(cx, cy);
   524→   524→      this.pushUndo();
   525→   525→      this.painting = true;
   526→   526→      this.lastTile = { x: tx, y: ty };
   527→   527→      this.applyToolAt(tx, ty);
   528→   528→    });
   529→   529→    this.on(display, 'pointermove', (e) => {
   530→   530→      const pe = e as PointerEvent;
   531→   531→      const rect = display.getBoundingClientRect();
   532→   532→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   533→   533→      const { tx, ty } = this.tileAt(cx, cy);
   534→   534→      this.hover = { tx, ty };
   535→   535→      this.hooks.onHover?.(tx, ty);
   536→   536→      if (this.panning) {
   537→   537→        this.cam.x -= (pe.clientX - this.lastPan.x) / this.zoom;
   538→   538→        this.cam.y -= (pe.clientY - this.lastPan.y) / this.zoom;
   539→   539→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   540→   540→        return;
   541→   541→      }
   542→   542→      if (this.painting) {
   543→   543→        const lt = this.lastTile;
   544→   544→        if (!lt) {
   545→   545→          this.lastTile = { x: tx, y: ty };
   546→   546→          this.dragMode = true;
   547→   547→          this.applyToolAt(tx, ty);
   548→   548→          this.dragMode = false;
   549→   549→          return;
   550→   550→        }
   551→   551→        const steps = Math.max(Math.abs(tx - lt.x), Math.abs(ty - lt.y));
   552→   552→        if (steps > 0) {
   553→   553→          this.dragMode = true;
   554→   554→          for (let s = 1; s <= steps; s++) {
   555→   555→            const ix = Math.round(lt.x + (tx - lt.x) * s / steps);
   556→   556→            const iy = Math.round(lt.y + (ty - lt.y) * s / steps);
   557→   557→            this.applyToolAt(ix, iy);
   558→   558→          }
   559→   559→          this.dragMode = false;
   560→   560→          this.lastTile = { x: tx, y: ty };
   561→   561→        }
   562→   562→      }
   563→   563→    });
   564→   564→    const endPointer = (): void => {
   565→   565→      this.painting = false;
   566→   566→      this.panning = false;
   567→   567→      this.lastTile = null;
   568→   568→    };
   569→   569→    this.on(display, 'pointerup', endPointer);
   570→   570→    this.on(display, 'pointercancel', endPointer);
   571→   571→    this.on(display, 'pointerleave', () => { this.hover = null; });
   572→   572→    this.on(display, 'wheel', (e) => {
   573→   573→      const we = e as WheelEvent;
   574→   574→      we.preventDefault();
   575→   575→      const rect = display.getBoundingClientRect();
   576→   576→      this.setZoom(
   577→   577→        ZOOMS[Math.max(0, Math.min(ZOOMS.length - 1, ZOOMS.indexOf(this.zoom) + (we.deltaY < 0 ? 1 : -1)))],
   578→   578→        { x: we.clientX - rect.left, y: we.clientY - rect.top },
   579→   579→      );
   580→   580→    }, { passive: false });
   581→   581→
   582→   582→    this.on(window, 'keydown', (e) => {
   583→   583→      const ke = e as KeyboardEvent;
   584→   584→      if (ke.target instanceof HTMLInputElement || ke.target instanceof HTMLTextAreaElement) return;
   585→   585→      this.keys.add(ke.code);
   586→   586→      if (ke.code === 'Space') { e.preventDefault(); this.center(); }
   587→   587→      if (ke.code === 'Equal' || ke.code === 'NumpadAdd') this.cycleZoom(1);
   588→   588→      if (ke.code === 'Minus' || ke.code === 'NumpadSubtract') this.cycleZoom(-1);
   589→   589→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyZ') {
   590→   590→        e.preventDefault();
   591→   591→        if (ke.shiftKey) this.redo(); else this.undo();
   592→   592→      }
   593→   593→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyY') {
   594→   594→        e.preventDefault();
   595→   595→        this.redo();
   596→   596→      }
   597→   597→      if (ke.code === 'Escape') this.setTool({ t: 'view' });
   598→   598→    });
   599→   599→    this.on(window, 'keyup', (e) => this.keys.delete((e as KeyboardEvent).code));
   600→   600→  }

   604→   604→  private applyToolAt(tx: number, ty: number): void {
   605→   605→    const t = this.tool;
   606→   606→    switch (t.t) {
   607→   607→      case 'terr':
   608→   608→        if (this.map.setTerr(tx, ty, t.k)) this.beeper.play('paint');
   609→   609→        break;
   610→   610→      case 'road':
   611→   611→        if (this.map.setRoad(tx, ty, true)) this.beeper.play('paint');
   612→   612→        break;
   613→   613→      case 'tree':
   614→   614→        if (this.map.plantTree(tx, ty)) this.beeper.play('paint');
   615→   615→        break;
   616→   616→      case 'doze':
   617→   617→        if (this.map.bulldoze(tx, ty)) this.beeper.play('delete');
   618→   618→        break;
   619→   619→      case 'build': {
   620→   620→        if (this.map.place(t.id, t.variant, tx, ty)) {
   621→   621→          this.beeper.play('place');
   622→   622→        } else if (!this.dragMode && this.map.inB(tx, ty)) {
   623→   623→          this.beeper.play('error');
   624→   624→        }
   625→   625→        break;
   626→   626→      }
   627→   627→    }
   628→   628→  }
