   139→   139→  /** 世界格角点 → 等距屏幕 px（低分辨率画布坐标） */
   140→   140→  private w2sx(wx: number, wy: number): number { return (wx - wy) * 16 - this.cam.x + this.lw / 2; }

   143→   143→  /** 鼠标 CSS 坐标 → 格坐标 */
   144→   144→  tileAt(cssX: number, cssY: number): { tx: number; ty: number } {
   145→   145→    const sx = cssX / this.zoom, sy = cssY / this.zoom;
   146→   146→    const ax = sx + this.cam.x - this.lw / 2;   // = (X−Y)*16

   149→   149→    const Y = ay / 16 - ax / 32;

   172→   172→  setZoom(z: number, focusCss?: { x: number; y: number }): void {
   173→   173→    if (z === this.zoom) return;
   174→   174→    let keep: { gx: number; gy: number } | null = null;
   175→   175→    if (focusCss) {
   176→   176→      const t = this.tileAt(focusCss.x, focusCss.y);
   177→   177→      keep = { gx: t.tx + 0.5, gy: t.ty + 0.5 };
   178→   178→    }
   179→   179→    this.zoom = z;
   180→   180→    this.resize();
   181→   181→    if (keep && focusCss) {
   182→   182→      const sx = focusCss.x / z, sy = focusCss.y / z;
   183→   183→      this.cam.x = (keep.gx - keep.gy) * 16 - sx + this.lw / 2;
   184→   184→      this.cam.y = (keep.gx + keep.gy) * 8 - sy + this.lh / 2;
   185→   185→    }
   186→   186→    this.beeper.play('zoom');
   187→   187→    this.hooks.onZoom?.(z);
   188→   188→  }

   192→   192→  private loop = (): void => {
   193→   193→    this.raf = requestAnimationFrame(this.loop);
   194→   194→    const now = performance.now();
   195→   195→    const dt = Math.min(100, now - this.lastT);
   196→   196→    this.lastT = now;
   197→   197→    this.fpsAcc += dt; this.fpsN++;
   198→   198→    if (now - this.fpsAt > 500) {
   199→   199→      this.hooks.onFps?.(Math.round(1000 / (this.fpsAcc / this.fpsN)));
   200→   200→      this.fpsAcc = 0; this.fpsN = 0; this.fpsAt = now;
   201→   201→    }
   202→   202→    const pan = dt * 0.72;
   203→   203→    if (this.keys.has('ArrowLeft') || this.keys.has('KeyA')) this.cam.x -= pan;
   204→   204→    if (this.keys.has('ArrowRight') || this.keys.has('KeyD')) this.cam.x += pan;
   205→   205→    if (this.keys.has('ArrowUp') || this.keys.has('KeyW')) this.cam.y -= pan;
   206→   206→    if (this.keys.has('ArrowDown') || this.keys.has('KeyS')) this.cam.y += pan;
   207→   207→    this.render(now);
   208→   208→    if (this.map.version !== this.lastSaveVer && now - this.lastSaveT > 25000) {
   209→   209→      this.autoSave();
   210→   210→    }
   211→   211→  };

   215→   215→  private visibleRange(): { x0: number; y0: number; x1: number; y1: number } {
   216→   216→    const n = this.map.n;
   217→   217→    const cs: Array<[number, number]> = [[0, 0], [this.lw, 0], [0, this.lh], [this.lw, this.lh]];
   218→   218→    let minX = n, maxX = 0, minY = n, maxY = 0;
   219→   219→    for (const [sx, sy] of cs) {
   220→   220→      const ax = sx + this.cam.x - this.lw / 2;
   221→   221→      const ay = sy + this.cam.y - this.lh / 2;
   222→   222→      const X = ax / 32 + ay / 16;
   223→   223→      const Y = ay / 16 - ax / 32;
   224→   224→      minX = Math.min(minX, Math.floor(X)); maxX = Math.max(maxX, Math.ceil(X));
   225→   225→      minY = Math.min(minY, Math.floor(Y)); maxY = Math.max(maxY, Math.ceil(Y));
   226→   226→    }
   227→   227→    const pad = 4;
   228→   228→    return {
   229→   229→      x0: Math.max(0, minX - pad), y0: Math.max(0, minY - pad),
   230→   230→      x1: Math.min(n - 1, maxX + pad), y1: Math.min(n - 1, maxY + pad),
   231→   231→    };
   232→   232→  }

   234→   234→  private render(t: number): void {
   235→   235→    const ctx = this.lctx;
   236→   236→    const { lw, lh } = this;
   237→   237→
   238→   238→    // 背景虚空（深海蓝 + 稀疏星点抖动）
   239→   239→    ctx.fillStyle = '#0B0F24';
   240→   240→    ctx.fillRect(0, 0, lw, lh);
   241→   241→    ctx.fillStyle = '#182142';
   242→   242→    for (let y = 0; y < lh; y += 8) {
   243→   243→      for (let x = ((y >> 3) % 2) * 8; x < lw; x += 16) ctx.fillRect(x + 3, y + 5, 1, 1);
   244→   244→    }
   245→   245→
   246→   246→    const vr = this.visibleRange();
   247→   247→    const wf = Math.floor(t / 620) % 2;
   248→   248→    const items: RenderItem[] = [];
   249→   249→    const m = this.map;
   250→   250→
   251→   251→    // ---- Pass 1：地面 ----
   252→   252→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   253→   253→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   254→   254→        const i = m.idx(tx, ty);
   255→   255→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   256→   256→        if (sx < -96 || sx > lw + 96 || sy < -160 || sy > lh + 96) continue;
   257→   257→
   258→   258→        const a = m.occ[i];
   259→   259→        let groundDrawn = false;
   260→   260→        if (a >= 0) {
   261→   261→          const o = m.objs.get(a)!;
   262→   262→          const d = defById(o.id);
   263→   263→          const isAnchor = o.ax === tx && o.ay === ty;
   264→   264→          if (d.ground) {
   265→   265→            if (isAnchor) {
   266→   266→              const g = getGround(d.ground, { variant: o.v });
   267→   267→              ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   268→   268→            }
   269→   269→            groundDrawn = true;   // 大地面精灵覆盖全部占地格
   270→   270→          }
   271→   271→        }
   272→   272→        if (groundDrawn) continue;
   273→   273→
   274→   274→        if (m.road[i]) {
   275→   275→          const g = getGround('road', { road: this.roadMask(tx, ty), variant: Math.floor(hash2(tx, ty, 5) * 4) });
   276→   276→          ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   277→   277→        } else {
   278→   278→          const k = m.terr[i];
   279→   279→          const v = Math.floor(hash2(tx, ty, 9) * 4);
   280→   280→          if (k === TERR_WATER) {
   281→   281→            const g = getGround('water', { edge: this.landEdge(tx, ty), variant: v, frame: wf });
   282→   282→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   283→   283→          } else if (k === TERR_SAND) {
   284→   284→            const g = getGround('sand', { variant: v });
   285→   285→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   286→   286→          } else if (k === TERR_FARM) {
   287→   287→            const g = getGround('farm', { variant: v, edge: this.waterEdge(tx, ty) });
   288→   288→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   289→   289→          } else if (k === TERR_ROCK) {
   290→   290→            const g = getGround('rock', { variant: v, edge: this.waterEdge(tx, ty) });
   291→   291→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   292→   292→          } else {
   293→   293→            const g = getGround('grass', { variant: v, edge: this.waterEdge(tx, ty) });
   294→   294→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   295→   295→          }
   296→   296→        }
   297→   297→      }
   298→   298→    }
   299→   299→
   300→   300→    // ---- Pass 2：树木 + 建筑（按前角深度排序）----
   301→   301→    const seen = new Set<number>();
   302→   302→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   303→   303→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   304→   304→        const i = m.idx(tx, ty);
   305→   305→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   306→   306→        if (sx < -160 || sx > lw + 160 || sy < -300 || sy > lh + 160) continue;
   307→   307→
   308→   308→        const tv = m.tree[i];
   309→   309→        if (tv >= 0 && !m.road[i] && m.occ[i] < 0) {
   310→   310→          const sp = this.treeSprites[tv & 3];
   311→   311→          const key = tx + ty;
   312→   312→          const dx = Math.round(sx + sp.ox), dy = Math.round(sy + sp.oy);
   313→   313→          items.push({ key, draw: () => ctx.drawImage(sp.cv, dx, dy) });
   314→   314→        }
   315→   315→
   316→   316→        const a = m.occ[i];
   317→   317→        if (a >= 0 && !seen.has(a)) {
   318→   318→          seen.add(a);
   319→   319→          const o = m.objs.get(a)!;
   320→   320→          const d = defById(o.id);
   321→   321→          const bs = this.book.get(o.id, o.v);
   322→   322→          const sp = bs.sp;
   323→   323→          const bsx = this.w2sx(o.ax, o.ay), bsy = this.w2sy(o.ax, o.ay);
   324→   324→          const key = (o.ax + d.w - 1) + (o.ay + d.h - 1);
   325→   325→          const blinkOn = Math.floor(t / 640) % 2 === 0;
   326→   326→          const smokeFr = Math.floor(t / 300) % 3;
   327→   327→          const bx = Math.round(bsx + sp.ox), by = Math.round(bsy + sp.oy);
   328→   328→          const smk = bs.smokeAt;
   329→   329→          const blk = bs.blinkAt;
   330→   330→          const sm = smk ? this.book.smoke[smokeFr] : null;
   331→   331→          const mx = smk ? bsx + (smk[0] - smk[1]) : 0;
   332→   332→          const my = smk ? bsy + (smk[0] + smk[1]) / 2 - smk[2] : 0;
   333→   333→          const lx = blk ? bsx + (blk[0] - blk[1]) : 0;
   334→   334→          const ly = blk ? bsy + (blk[0] + blk[1]) / 2 - blk[2] : 0;
   335→   335→          items.push({
   336→   336→            key,
   337→   337→            draw: () => {
   338→   338→              ctx.drawImage(sp.cv, bx, by);
   339→   339→              if (sm) ctx.drawImage(sm.cv, Math.round(mx + sm.ox), Math.round(my + sm.oy));
   340→   340→              if (blk && blinkOn) {
   341→   341→                ctx.fillStyle = PALETTE[62];
   342→   342→                ctx.fillRect(Math.round(lx) - 1, Math.round(ly) - 1, 2, 2);
   343→   343→                ctx.fillStyle = PALETTE[61];
   344→   344→                ctx.fillRect(Math.round(lx), Math.round(ly), 1, 1);
   345→   345→              }
   346→   346→            },
   347→   347→          });
   348→   348→        }
   349→   349→      }
   350→   350→    }
   351→   351→    items.sort((p, q) => p.key - q.key);
   352→   352→    for (const it of items) it.draw();
   353→   353→
   354→   354→    // ---- Pass 3：光标 + 幽灵 ----
   355→   355→    this.drawCursor(ctx);
   356→   356→
   357→   357→    // ---- 上屏（最近邻放大 + 扫描线）----
   358→   358→    const d = this.display;
   359→   359→    const dctx = this.dctx;
   360→   360→    if (d && dctx) {
   361→   361→      const S = this.zoom * this.dpr;
   362→   362→      dctx.imageSmoothingEnabled = false;
   363→   363→      dctx.drawImage(this.lowres, 0, 0, lw, lh, 0, 0, Math.round(lw * S), Math.round(lh * S));
   364→   364→      const step = Math.max(2, Math.round(S));
   365→   365→      dctx.globalAlpha = 0.15;
   366→   366→      dctx.fillStyle = '#000';
   367→   367→      for (let y = step - 1; y < d.height; y += step) dctx.fillRect(0, y, d.width, 1);
   368→   368→      dctx.globalAlpha = 1;
   369→   369→    }
   370→   370→
   371→   371→    this.renderMini();
   372→   372→  }
   373→   373→
   374→   374→  private roadMask(tx: number, ty: number): number {
   375→   375→    const m = this.map;
   376→   376→    let v = 0;
   377→   377→    if (m.inB(tx, ty - 1) && m.road[m.idx(tx, ty - 1)]) v |= 1;
   378→   378→    if (m.inB(tx + 1, ty) && m.road[m.idx(tx + 1, ty)]) v |= 2;
   379→   379→    if (m.inB(tx, ty + 1) && m.road[m.idx(tx, ty + 1)]) v |= 4;
   380→   380→    if (m.inB(tx - 1, ty) && m.road[m.idx(tx - 1, ty)]) v |= 8;
   381→   381→    return v;
   382→   382→  }
   383→   383→
   384→   384→  private waterEdge(tx: number, ty: number): number {
   385→   385→    const m = this.map;
   386→   386→    let v = 0;
   387→   387→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] === TERR_WATER) v |= 1;
   388→   388→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] === TERR_WATER) v |= 2;
   389→   389→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] === TERR_WATER) v |= 4;
   390→   390→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] === TERR_WATER) v |= 8;
   391→   391→    return v;
   392→   392→  }

   394→   394→  private landEdge(tx: number, ty: number): number {
   395→   395→    const m = this.map;
   396→   396→    let v = 0;
   397→   397→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] !== TERR_WATER) v |= 1;
   398→   398→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] !== TERR_WATER) v |= 2;
   399→   399→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] !== TERR_WATER) v |= 4;
   400→   400→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] !== TERR_WATER) v |= 8;
   401→   401→    return v;
   402→   402→  }

   404→   404→  /** 屏幕偏移是否在 w×h 足印菱形内（相对锚点） */
   405→   405→  private inFootprint(dx: number, dy: number, w: number, h: number): boolean {
   406→   406→    const X = dx / 2 + dy;
   407→   407→    const Y = dy - dx / 2;
   408→   408→    return X >= 0 && X <= 16 * w && Y >= 0 && Y <= 16 * h;
   409→   409→  }

   411→   411→  /** 光标菱形 + 放置幽灵 */
   412→   412→  private drawCursor(ctx: CanvasRenderingContext2D): void {
   413→   413→    const h = this.hover;
   414→   414→    if (!h || !this.map.inB(h.tx, h.ty)) return;
   415→   415→    const tool = this.tool;
   416→   416→    let w = 1, hh = 1;
   417→   417→    if (tool.t === 'build') {
   418→   418→      const d = defById(tool.id);
   419→   419→      w = d.w; hh = d.h;
   420→   420→    }
   421→   421→    const x0 = this.w2sx(h.tx, h.ty), y0 = this.w2sy(h.tx, h.ty);
   422→   422→    ctx.beginPath();
   423→   423→    ctx.moveTo(x0, y0);
   424→   424→    ctx.lineTo(x0 + 16 * w, y0 + 8 * w);
   425→   425→    ctx.lineTo(x0 + 16 * (w - hh), y0 + 8 * (w + hh));
   426→   426→    ctx.lineTo(x0 - 16 * hh, y0 + 8 * hh);
   427→   427→    ctx.closePath();
   428→   428→    ctx.strokeStyle = 'rgba(8,10,18,0.85)';
   429→   429→    ctx.lineWidth = 3;
   430→   430→    ctx.stroke();
   431→   431→    ctx.strokeStyle = '#F4F2E8';
   432→   432→    ctx.lineWidth = 1;
   433→   433→    ctx.stroke();
   434→   434→
   435→   435→    if (tool.t === 'build') {
   436→   436→      const bs = this.book.get(tool.id, tool.variant);
   437→   437→      const g = ghostSprite(bs.sp);
   438→   438→      const ok = this.map.canPlace(tool.id, h.tx, h.ty);
   439→   439→      ctx.drawImage(g, Math.round(x0 + bs.sp.ox), Math.round(y0 + bs.sp.oy));
   440→   440→      if (!ok) {
   441→   441→        ctx.fillStyle = 'rgba(224,85,72,0.55)';
   442→   442→        const W = 16 * (w + hh), H = 8 * (w + hh) + 2;
   443→   443→        for (let yy = 0; yy < H; yy += 2) {
   444→   444→          for (let xx = ((yy >> 1) % 2) * 2; xx < W * 2; xx += 4) {
   445→   445→            const px = x0 - 16 * hh + xx, py = y0 + yy;
   446→   446→            if (this.inFootprint(px - x0, py - y0, w, hh)) ctx.fillRect(px, py, 2, 1);
   447→   447→          }
   448→   448→        }
   449→   449→      }
   450→   450→    }
   451→   451→  }

   455→   455→  private renderMini(): void {
   456→   456→    const mini = this.mini;
   457→   457→    if (!mini) return;
   458→   458→    const mctx = mini.getContext('2d')!;
   459→   459→    const n = this.map.n;
   460→   460→    if (this.miniDirty) {
   461→   461→      const W = n * MM;
   462→   462→      if (this.miniBase.width !== W) { this.miniBase.width = W; this.miniBase.height = W; }
   463→   463→      const bctx = this.miniBase.getContext('2d')!;
   464→   464→      const img = bctx.createImageData(W, W);
   465→   465→      const u32 = new Uint32Array(img.data.buffer);
   466→   466→      const abgr = (hex: string): number => {
   467→   467→        const q = parseInt(hex.slice(1), 16);
   468→   468→        return (255 << 24) | ((q & 255) << 16) | (((q >> 8) & 255) << 8) | ((q >> 16) & 255);
   469→   469→      };
   470→   470→      const C = {
   471→   471→        grass: abgr('#3B7138'), water: abgr('#3467AC'), sand: abgr('#CFB379'),
   472→   472→        farm: abgr('#B9C46B'), rock: abgr('#6E737F'), tree: abgr('#245129'),
   473→   473→        road: abgr('#4A4C55'),
   474→   474→        res: abgr('#C25B44'), com: abgr('#E8CC4C'), ind: abgr('#9BA0A7'), civ: abgr('#5BC8B8'),
   475→   475→      };
   476→   476→      for (let y = 0; y < n; y++) {
   477→   477→        for (let x = 0; x < n; x++) {
   478→   478→          const i = this.map.idx(x, y);
   479→   479→          const k = this.map.terr[i];
   480→   480→          let cv: number = k === 1 ? C.water : k === 2 ? C.sand
   481→   481→            : k === 3 ? C.farm : k === 4 ? C.rock : C.grass;
   482→   482→          if (this.map.tree[i] >= 0) cv = C.tree;
   483→   483→          if (this.map.road[i]) cv = C.road;
   484→   484→          const a = this.map.occ[i];
   485→   485→          if (a >= 0) {
   486→   486→            const o = this.map.objs.get(a)!;
   487→   487→            const cat = defById(o.id).cat;
   488→   488→            cv = cat === 'res' ? C.res : cat === 'com' ? C.com
   489→   489→              : cat === 'ind' ? C.ind : C.civ;
   490→   490→          }
   491→   491→          for (let dy = 0; dy < MM; dy++) {
   492→   492→            const row = (y * MM + dy) * W + x * MM;
   493→   493→            for (let dx = 0; dx < MM; dx++) u32[row + dx] = cv;
   494→   494→          }
   495→   495→        }
   496→   496→      }
   497→   497→      bctx.putImageData(img, 0, 0);
   498→   498→      this.miniDirty = false;
   499→   499→    }
   500→   500→    mctx.imageSmoothingEnabled = false;
   501→   501→    mctx.drawImage(this.miniBase, 0, 0);
   502→   502→    const vr = this.visibleRange();
   503→   503→    mctx.strokeStyle = '#F4F2E8';
   504→   504→    mctx.lineWidth = 1;
   505→   505→    mctx.strokeRect(vr.x0 * MM + 0.5, vr.y0 * MM + 0.5, (vr.x1 - vr.x0) * MM, (vr.y1 - vr.y0) * MM);
   506→   506→  }

   508→   508→  private bindMini(mini: HTMLCanvasElement): void {
   509→   509→    const jump = (e: PointerEvent): void => {
   510→   510→      const rect = mini.getBoundingClientRect();
   511→   511→      const tx = Math.floor(((e.clientX - rect.left) / rect.width) * this.map.n);
   512→   512→      const ty = Math.floor(((e.clientY - rect.top) / rect.height) * this.map.n);
   513→   513→      this.cam.x = (tx - ty) * 16;
   514→   514→      this.cam.y = (tx + ty) * 8;

   519→   519→      const up = () => {
   520→   520→        mini.removeEventListener('pointermove', mv);
   521→   521→        mini.removeEventListener('pointerup', up);
   522→   522→      };
   523→   523→      mini.addEventListener('pointermove', mv);
   524→   524→      mini.addEventListener('pointerup', up);
   525→   525→    });
   526→   526→  }

   530→   530→  private bindInput(display: HTMLCanvasElement): void {
   531→   531→    this.on(display, 'contextmenu', (e) => e.preventDefault());
   532→   532→    this.on(display, 'pointerdown', (e) => {
   533→   533→      const pe = e as PointerEvent;
   534→   534→      display.setPointerCapture(pe.pointerId);
   535→   535→      const rect = display.getBoundingClientRect();
   536→   536→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   537→   537→      this.beeper.play('click');
   538→   538→      if (pe.button === 1 || pe.button === 2 || this.tool.t === 'view') {
   539→   539→        this.panning = true;
   540→   540→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   541→   541→        return;
   542→   542→      }
   543→   543→      if (pe.button !== 0) return;
   544→   544→      const { tx, ty } = this.tileAt(cx, cy);
   545→   545→      this.pushUndo();
   546→   546→      this.painting = true;
   547→   547→      this.lastTile = { x: tx, y: ty };
   548→   548→      this.applyToolAt(tx, ty);
   549→   549→    });
   550→   550→    this.on(display, 'pointermove', (e) => {
   551→   551→      const pe = e as PointerEvent;
   552→   552→      const rect = display.getBoundingClientRect();
   553→   553→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   554→   554→      const { tx, ty } = this.tileAt(cx, cy);
   555→   555→      this.hover = { tx, ty };
   556→   556→      this.hooks.onHover?.(tx, ty);
   557→   557→      if (this.panning) {
   558→   558→        this.cam.x -= (pe.clientX - this.lastPan.x) / this.zoom;
   559→   559→        this.cam.y -= (pe.clientY - this.lastPan.y) / this.zoom;
   560→   560→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   561→   561→        return;
   562→   562→      }
   563→   563→      if (this.painting) {
   564→   564→        const lt = this.lastTile;
   565→   565→        if (!lt) {
   566→   566→          this.lastTile = { x: tx, y: ty };
   567→   567→          this.dragMode = true;
   568→   568→          this.applyToolAt(tx, ty);
   569→   569→          this.dragMode = false;
   570→   570→          return;
   571→   571→        }
   572→   572→        const steps = Math.max(Math.abs(tx - lt.x), Math.abs(ty - lt.y));
   573→   573→        if (steps > 0) {
   574→   574→          this.dragMode = true;
   575→   575→          for (let s = 1; s <= steps; s++) {
   576→   576→            const ix = Math.round(lt.x + (tx - lt.x) * s / steps);
   577→   577→            const iy = Math.round(lt.y + (ty - lt.y) * s / steps);
   578→   578→            this.applyToolAt(ix, iy);
   579→   579→          }
   580→   580→          this.dragMode = false;
   581→   581→          this.lastTile = { x: tx, y: ty };
   582→   582→        }
   583→   583→      }
   584→   584→    });
   585→   585→    const endPointer = (): void => {
   586→   586→      this.painting = false;
   587→   587→      this.panning = false;
   588→   588→      this.lastTile = null;
   589→   589→    };
   590→   590→    this.on(display, 'pointerup', endPointer);
   591→   591→    this.on(display, 'pointercancel', endPointer);
   592→   592→    this.on(display, 'pointerleave', () => { this.hover = null; });
   593→   593→    this.on(display, 'wheel', (e) => {
   594→   594→      const we = e as WheelEvent;
   595→   595→      we.preventDefault();
   596→   596→      const rect = display.getBoundingClientRect();
   597→   597→      this.setZoom(
   598→   598→        ZOOMS[Math.max(0, Math.min(ZOOMS.length - 1, ZOOMS.indexOf(this.zoom) + (we.deltaY < 0 ? 1 : -1)))],
   599→   599→        { x: we.clientX - rect.left, y: we.clientY - rect.top },
   600→   600→      );
   601→   601→    }, { passive: false });
   602→   602→
   603→   603→    this.on(window, 'keydown', (e) => {
   604→   604→      const ke = e as KeyboardEvent;
   605→   605→      if (ke.target instanceof HTMLInputElement || ke.target instanceof HTMLTextAreaElement) return;
   606→   606→      this.keys.add(ke.code);
   607→   607→      if (ke.code === 'Space') { e.preventDefault(); this.center(); }
   608→   608→      if (ke.code === 'Equal' || ke.code === 'NumpadAdd') this.cycleZoom(1);
   609→   609→      if (ke.code === 'Minus' || ke.code === 'NumpadSubtract') this.cycleZoom(-1);
   610→   610→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyZ') {
   611→   611→        e.preventDefault();
   612→   612→        if (ke.shiftKey) this.redo(); else this.undo();
   613→   613→      }
   614→   614→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyY') {
   615→   615→        e.preventDefault();
   616→   616→        this.redo();
   617→   617→      }
   618→   618→      if (ke.code === 'Escape') this.setTool({ t: 'view' });
   619→   619→    });
   620→   620→    this.on(window, 'keyup', (e) => this.keys.delete((e as KeyboardEvent).code));
   621→   621→  }

   625→   625→  private applyToolAt(tx: number, ty: number): void {
   626→   626→    const t = this.tool;
   627→   627→    switch (t.t) {
   628→   628→      case 'terr':
   629→   629→        if (this.map.setTerr(tx, ty, t.k)) this.beeper.play('paint');
   630→   630→        break;
   631→   631→      case 'road':
   632→   632→        if (this.map.setRoad(tx, ty, true)) this.beeper.play('paint');
   633→   633→        break;
   634→   634→      case 'tree':
   635→   635→        if (this.map.plantTree(tx, ty)) this.beeper.play('paint');
   636→   636→        break;
   637→   637→      case 'doze':
   638→   638→        if (this.map.bulldoze(tx, ty)) this.beeper.play('delete');
   639→   639→        break;
   640→   640→      case 'build': {
   641→   641→        if (this.map.place(t.id, t.variant, tx, ty)) {
   642→   642→          this.beeper.play('place');
   643→   643→        } else if (!this.dragMode && this.map.inB(tx, ty)) {
   644→   644→          this.beeper.play('error');
   645→   645→        }
   646→   646→        break;
   647→   647→      }
   648→   648→    }
