   102→  async init(onProgress: (frac: number, label: string) => void): Promise<void> {
   103→    await this.book.init(onProgress);
   104→    this.treeSprites = [0, 1, 2, 3, 4, 5].map(v => renderModel(genTree(v).model));

   110→  attach(display: HTMLCanvasElement, mini: HTMLCanvasElement | null): void {
   111→    this.display = display;
   112→    this.dctx = display.getContext('2d');
   113→    this.mini = mini;
   114→    if (mini) {
   115→      mini.width = this.map.n * MM;
   116→      mini.height = this.map.n * MM;
   117→      this.bindMini(mini);
   118→    }
   119→    this.bindInput(display);
   120→    // 移动端 / 触屏默认 ×1（屏幕窄，×2 太近）
   121→    const coarse = window.matchMedia?.('(pointer: coarse)').matches ?? false;
   122→    if (coarse || Math.min(window.innerWidth, window.innerHeight) < 500) {
   123→      this.setZoom(1);
   124→    }
   125→    this.resize();
   126→    this.loop();
   127→  }

   129→  destroy(): void {
   130→    cancelAnimationFrame(this.raf);
   131→    for (const [t, n, f] of this.listeners) t.removeEventListener(n, f);
   132→    this.listeners = [];
   133→  }

   135→  private on(t: EventTarget, name: string, f: EventListener, opts?: AddEventListenerOptions): void {
   136→    t.addEventListener(name, f, opts);
   137→    this.listeners.push([t, name, f]);
   138→  }
   139→
   140→  resize(): void {
   141→    const d = this.display;
   142→    if (!d) return;
   143→    const rect = d.getBoundingClientRect();
   144→    this.dpr = Math.min(2, window.devicePixelRatio || 1);
   145→    const cssW = Math.max(100, rect.width);
   146→    const cssH = Math.max(100, rect.height);
   147→    d.width = Math.round(cssW * this.dpr);
   148→    d.height = Math.round(cssH * this.dpr);
   149→    this.lowres.width = Math.ceil(cssW / this.zoom);
   150→    this.lowres.height = Math.ceil(cssH / this.zoom);
   151→    this.lctx.imageSmoothingEnabled = false;
   152→    if (this.dctx) this.dctx.imageSmoothingEnabled = false;
   153→  }

   155→  // ---------------- 坐标 ----------------
   156→
   157→  private get lw(): number { return this.lowres.width; }
   158→  private get lh(): number { return this.lowres.height; }
   159→
   160→  /** 世界格角点 → 等距屏幕 px（低分辨率画布坐标） */
   161→  private w2sx(wx: number, wy: number): number { return (wx - wy) * 16 - this.cam.x + this.lw / 2; }
   162→  private w2sy(wx: number, wy: number): number { return (wx + wy) * 8 - this.cam.y + this.lh / 2; }
   163→
   164→  /** 鼠标 CSS 坐标 → 格坐标 */
   165→  tileAt(cssX: number, cssY: number): { tx: number; ty: number } {
   166→    const sx = cssX / this.zoom, sy = cssY / this.zoom;
   167→    const ax = sx + this.cam.x - this.lw / 2;   // = (X−Y)*16
   168→    const ay = sy + this.cam.y - this.lh / 2;   // = (X+Y)*8
   169→    const X = ax / 32 + ay / 16;
   170→    const Y = ay / 16 - ax / 32;
   171→    return { tx: Math.floor(X), ty: Math.floor(Y) };
   172→  }
   173→
   174→  center(): void {
   175→    const c = this.map.n / 2;
   176→    this.cam.x = (c - c) * 16;
   177→    this.cam.y = (c + c) * 8;
   178→  }

   204→  cycleZoom(dir: number): void {
   205→    const i = ZOOMS.indexOf(this.zoom);
   206→    const j = Math.max(0, Math.min(ZOOMS.length - 1, i + dir));
   207→    if (j !== i) this.setZoom(ZOOMS[j]);
   208→  }
   209→
   210→  setZoom(z: number, focusCss?: { x: number; y: number }): void {
   211→    if (z === this.zoom) return;
   212→    let keep: { gx: number; gy: number } | null = null;
   213→    if (focusCss) {
   214→      const t = this.tileAt(focusCss.x, focusCss.y);
   215→      keep = { gx: t.tx + 0.5, gy: t.ty + 0.5 };
   216→    }
   217→    this.zoom = z;
   218→    this.resize();
   219→    if (keep && focusCss) {
   220→      const sx = focusCss.x / z, sy = focusCss.y / z;
   221→      this.cam.x = (keep.gx - keep.gy) * 16 - sx + this.lw / 2;
   222→      this.cam.y = (keep.gx + keep.gy) * 8 - sy + this.lh / 2;
   223→    }
   224→    this.beeper.play('zoom');
   225→    this.hooks.onZoom?.(z);
   226→  }
   227→
   228→  // ---------------- 主循环 ----------------
   229→
   230→  private loop = (): void => {
   231→    this.raf = requestAnimationFrame(this.loop);
   232→    const now = performance.now();
   233→    const dt = Math.min(100, now - this.lastT);
   234→    this.lastT = now;
   235→    this.fpsAcc += dt; this.fpsN++;
   236→    if (now - this.fpsAt > 500) {
   237→      this.hooks.onFps?.(Math.round(1000 / (this.fpsAcc / this.fpsN)));
   238→      this.fpsAcc = 0; this.fpsN = 0; this.fpsAt = now;
   239→    }
   240→    const pan = dt * 0.72;
   241→    if (this.keys.has('ArrowLeft') || this.keys.has('KeyA')) this.cam.x -= pan;
   242→    if (this.keys.has('ArrowRight') || this.keys.has('KeyD')) this.cam.x += pan;
   243→    if (this.keys.has('ArrowUp') || this.keys.has('KeyW')) this.cam.y -= pan;
   244→    if (this.keys.has('ArrowDown') || this.keys.has('KeyS')) this.cam.y += pan;
   245→    if (this.crowds) this.actors.tick(dt / 1000, this.map);
   246→    this.render(now);
   247→    if (this.map.version !== this.lastSaveVer && now - this.lastSaveT > 25000) {
   248→      this.autoSave();
   249→    }
   250→  };

   254→  private visibleRange(): { x0: number; y0: number; x1: number; y1: number } {
   255→    const n = this.map.n;
   256→    const cs: Array<[number, number]> = [[0, 0], [this.lw, 0], [0, this.lh], [this.lw, this.lh]];
   257→    let minX = n, maxX = 0, minY = n, maxY = 0;
   258→    for (const [sx, sy] of cs) {
   259→      const ax = sx + this.cam.x - this.lw / 2;
   260→      const ay = sy + this.cam.y - this.lh / 2;
   261→      const X = ax / 32 + ay / 16;
   262→      const Y = ay / 16 - ax / 32;
   263→      minX = Math.min(minX, Math.floor(X)); maxX = Math.max(maxX, Math.ceil(X));
   264→      minY = Math.min(minY, Math.floor(Y)); maxY = Math.max(maxY, Math.ceil(Y));
   265→    }
   266→    const pad = 4;
   267→    return {
   268→      x0: Math.max(0, minX - pad), y0: Math.max(0, minY - pad),
   269→      x1: Math.min(n - 1, maxX + pad), y1: Math.min(n - 1, maxY + pad),
   270→    };
   271→  }

   273→  private render(t: number): void {
   274→    const ctx = this.lctx;
   275→    const { lw, lh } = this;
   276→
   277→    // 背景虚空（深海蓝 + 稀疏星点抖动）
   278→    ctx.fillStyle = '#0B0F24';
   279→    ctx.fillRect(0, 0, lw, lh);
   280→    ctx.fillStyle = '#182142';
   281→    for (let y = 0; y < lh; y += 8) {
   282→      for (let x = ((y >> 3) % 2) * 8; x < lw; x += 16) ctx.fillRect(x + 3, y + 5, 1, 1);
   283→    }
   284→
   285→    const vr = this.visibleRange();
   286→    const wf = Math.floor(t / 620) % 2;
   287→    const items: RenderItem[] = [];
   288→    const m = this.map;
   289→
   290→    // ---- Pass 1：地面 ----
   291→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   292→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   293→        const i = m.idx(tx, ty);
   294→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   295→        if (sx < -96 || sx > lw + 96 || sy < -160 || sy > lh + 96) continue;
   296→
   297→        const a = m.occ[i];
   298→        let groundDrawn = false;
   299→        if (a >= 0) {
   300→          const o = m.objs.get(a)!;
   301→          const d = defById(o.id);
   302→          const isAnchor = o.ax === tx && o.ay === ty;
   303→          if (d.ground) {
   304→            if (isAnchor) {
   305→              const g = getGround(d.ground, { variant: o.v });
   306→              ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   307→            }
   308→            groundDrawn = true;   // 大地面精灵覆盖全部占地格
   309→          }
   310→        }
   311→        if (groundDrawn) continue;
   312→
   313→        if (m.road[i]) {
   314→          const g = getGround('road', { road: this.roadMask(tx, ty), variant: Math.floor(hash2(tx, ty, 5) * 4) });
   315→          ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   316→        } else {
   317→          const k = m.terr[i];
   318→          const v = Math.floor(hash2(tx, ty, 9) * 4);
   319→          const edge = this.waterEdge(tx, ty);
   320→          if (k === TERR_WATER) {
   321→            const g = getGround('water', { edge: this.landEdge(tx, ty), variant: v, frame: wf });
   322→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   323→          } else if (k === TERR_SAND) {
   324→            const g = getGround('sand', { variant: v });
   325→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   326→          } else if (k === TERR_FARM) {
   327→            const g = getGround('farm', { variant: v, edge });
   328→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   329→          } else if (k === TERR_ROCK) {
   330→            const g = getGround('rock', { variant: v, edge });
   331→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   332→          } else if (k === TERR_SNOW) {
   333→            const g = getGround('snow', { variant: v, edge });
   334→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   335→          } else if (k === TERR_FLOWER) {
   336→            const g = getGround('flower', { variant: v, edge });
   337→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   338→          } else if (k === TERR_DRY) {
   339→            const g = getGround('dry', { variant: v, edge });
   340→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   341→          } else if (k === TERR_SWAMP) {
   342→            const g = getGround('swamp', { variant: v, edge });
   343→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   344→          } else if (k === TERR_AUTUMN) {
   345→            const g = getGround('autumn', { variant: v, edge });
   346→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   347→          } else {
   348→            const g = getGround('grass', { variant: v, edge });
   349→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   350→          }
   351→        }
   352→      }
   353→    }
   354→
   355→    // ---- Pass 2：树木 + 建筑（按前角深度排序）----
   356→    const seen = new Set<number>();
   357→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   358→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   359→        const i = m.idx(tx, ty);
   360→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   361→        if (sx < -160 || sx > lw + 160 || sy < -300 || sy > lh + 160) continue;
   362→
   363→        const tv = m.tree[i];
   364→        if (tv >= 0 && !m.road[i] && m.occ[i] < 0) {
   365→          const sp = this.treeSprites[tv % this.treeSprites.length];
   366→          const key = tx + ty;
   367→          const dx = Math.round(sx + sp.ox), dy = Math.round(sy + sp.oy);
   368→          items.push({ key, draw: () => ctx.drawImage(sp.cv, dx, dy) });
   369→        }
   370→
   371→        const a = m.occ[i];
   372→        if (a >= 0 && !seen.has(a)) {
   373→          seen.add(a);
   374→          const o = m.objs.get(a)!;
   375→          const d = defById(o.id);
   376→          const bs = this.book.get(o.id, o.v);
   377→          const sp = bs.sp;
   378→          const bsx = this.w2sx(o.ax, o.ay), bsy = this.w2sy(o.ax, o.ay);
   379→          const key = (o.ax + d.w - 1) + (o.ay + d.h - 1);
   380→          const blinkOn = Math.floor(t / 640) % 2 === 0;
   381→          const smokeFr = Math.floor(t / 300) % 3;
   382→          const bx = Math.round(bsx + sp.ox), by = Math.round(bsy + sp.oy);
   383→          const smk = bs.smokeAt;
   384→          const blk = bs.blinkAt;
   385→          const sm = smk ? this.book.smoke[smokeFr] : null;
   386→          const mx = smk ? bsx + (smk[0] - smk[1]) : 0;
   387→          const my = smk ? bsy + (smk[0] + smk[1]) / 2 - smk[2] : 0;
   388→          const lx = blk ? bsx + (blk[0] - blk[1]) : 0;
   389→          const ly = blk ? bsy + (blk[0] + blk[1]) / 2 - blk[2] : 0;
   390→          items.push({
   391→            key,
   392→            draw: () => {
   393→              ctx.drawImage(sp.cv, bx, by);
   394→              if (sm) ctx.drawImage(sm.cv, Math.round(mx + sm.ox), Math.round(my + sm.oy));
   395→              if (blk && blinkOn) {
   396→                ctx.fillStyle = PALETTE[62];
   397→                ctx.fillRect(Math.round(lx) - 1, Math.round(ly) - 1, 2, 2);
   398→                ctx.fillStyle = PALETTE[61];
   399→                ctx.fillRect(Math.round(lx), Math.round(ly), 1, 1);
   400→              }
   401→            },
   402→          });
   403→        }
   404→      }
   405→    }
   406→    // ---- 行人 / 车辆（与建筑同序参与深度排序）----
   407→    if (this.crowds) {
   408→      for (const a of this.actors.actors) {
   409→        const px = a.fx + a.dx * a.t, py = a.fy + a.dy * a.t;
   410→        let wx: number, wy: number;
   411→        if (a.kind === 0) {
   412→          wx = (px + 0.5) * 16 + a.offX;
   413→          wy = (py + 0.5) * 16 + a.offY;
   414→        } else {
   415→          wx = px * 16 - a.dy * 2.3;
   416→          wy = py * 16 + a.dx * 2.3;
   417→        }
   418→        const sx = (wx - wy) - this.cam.x + lw / 2;
   419→        const sy = (wx + wy) / 2 - this.cam.y + lh / 2;
   420→        if (sx < -80 || sx > lw + 80 || sy < -60 || sy > lh + 60) continue;
   421→        const key = (wx + wy) / 16;
   422→        if (a.kind === 0) {
   423→          const d = ActorSystem.dirOf(a);
   424→          const fr = Math.floor(t / 220 + a.phase) % 2;
   425→          const cv = this.actors.personCv(d as 'dr' | 'dl' | 'ur' | 'ul', fr, a.look);
   426→          if (!cv) continue;
   427→          const dx = Math.round(sx - 3), dy = Math.round(sy - 12);
   428→          items.push({ key, draw: () => ctx.drawImage(cv, dx, dy) });
   429→        } else {
   430→          const d = ActorSystem.dirOf(a);
   431→          const sp = this.actors.carSp(d as 'e' | 'w' | 's' | 'n', a.look);
   432→          if (!sp) continue;
   433→          const dx = Math.round(sx + sp.ox), dy = Math.round(sy + sp.oy);
   434→          items.push({ key, draw: () => ctx.drawImage(sp.cv, dx, dy) });
   435→        }
   436→      }
   437→    }
   438→
   439→    items.sort((p, q) => p.key - q.key);
   440→    for (const it of items) it.draw();
   441→
   442→    // ---- Pass 3：光标 + 幽灵 ----
   443→    this.drawCursor(ctx);
   444→
   445→    // ---- 上屏（最近邻放大 + 扫描线）----
   446→    const d = this.display;
   447→    const dctx = this.dctx;
   448→    if (d && dctx) {
   449→      const S = this.zoom * this.dpr;
   450→      dctx.imageSmoothingEnabled = false;
   451→      dctx.drawImage(this.lowres, 0, 0, lw, lh, 0, 0, Math.round(lw * S), Math.round(lh * S));
   452→      const step = Math.max(2, Math.round(S));
   453→      dctx.globalAlpha = 0.15;
   454→      dctx.fillStyle = '#000';
   455→      for (let y = step - 1; y < d.height; y += step) dctx.fillRect(0, y, d.width, 1);
   456→      dctx.globalAlpha = 1;
   457→    }
   458→
   459→    this.renderMini();
   460→  }

   462→  private roadMask(tx: number, ty: number): number {
   463→    const m = this.map;
   464→    let v = 0;
   465→    if (m.inB(tx, ty - 1) && m.road[m.idx(tx, ty - 1)]) v |= 1;
   466→    if (m.inB(tx + 1, ty) && m.road[m.idx(tx + 1, ty)]) v |= 2;
   467→    if (m.inB(tx, ty + 1) && m.road[m.idx(tx, ty + 1)]) v |= 4;
   468→    if (m.inB(tx - 1, ty) && m.road[m.idx(tx - 1, ty)]) v |= 8;
   469→    return v;
   470→  }

   472→  private waterEdge(tx: number, ty: number): number {
   473→    const m = this.map;
   474→    let v = 0;
   475→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] === TERR_WATER) v |= 1;
   476→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] === TERR_WATER) v |= 2;
   477→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] === TERR_WATER) v |= 4;
   478→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] === TERR_WATER) v |= 8;
   479→    return v;
   480→  }

   482→  private landEdge(tx: number, ty: number): number {
   483→    const m = this.map;
   484→    let v = 0;
   485→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] !== TERR_WATER) v |= 1;
   486→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] !== TERR_WATER) v |= 2;
   487→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] !== TERR_WATER) v |= 4;
   488→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] !== TERR_WATER) v |= 8;
   489→    return v;

   492→  /** 屏幕偏移是否在 w×h 足印菱形内（相对锚点） */
   493→  private inFootprint(dx: number, dy: number, w: number, h: number): boolean {
   494→    const X = dx / 2 + dy;
   495→    const Y = dy - dx / 2;
   496→    return X >= 0 && X <= 16 * w && Y >= 0 && Y <= 16 * h;
   497→  }

   499→  /** 光标菱形 + 放置幽灵 */
   500→  private drawCursor(ctx: CanvasRenderingContext2D): void {
   501→    const h = this.hover;
   502→    if (!h || !this.map.inB(h.tx, h.ty)) return;
   503→    const tool = this.tool;
   504→    let w = 1, hh = 1;
   505→    if (tool.t === 'build') {
   506→      const d = defById(tool.id);
   507→      w = d.w; hh = d.h;
   508→    }
   509→    const x0 = this.w2sx(h.tx, h.ty), y0 = this.w2sy(h.tx, h.ty);
   510→    ctx.beginPath();
   511→    ctx.moveTo(x0, y0);
   512→    ctx.lineTo(x0 + 16 * w, y0 + 8 * w);
   513→    ctx.lineTo(x0 + 16 * (w - hh), y0 + 8 * (w + hh));
   514→    ctx.lineTo(x0 - 16 * hh, y0 + 8 * hh);
   515→    ctx.closePath();
   516→    ctx.strokeStyle = 'rgba(8,10,18,0.85)';
   517→    ctx.lineWidth = 3;
   518→    ctx.stroke();
   519→    ctx.strokeStyle = '#F4F2E8';
   520→    ctx.lineWidth = 1;
   521→    ctx.stroke();
   522→
   523→    if (tool.t === 'build') {
   524→      const bs = this.book.get(tool.id, tool.variant);
   525→      const g = ghostSprite(bs.sp);
   526→      const ok = this.map.canPlace(tool.id, h.tx, h.ty);
   527→      ctx.drawImage(g, Math.round(x0 + bs.sp.ox), Math.round(y0 + bs.sp.oy));
   528→      if (!ok) {
   529→        ctx.fillStyle = 'rgba(224,85,72,0.55)';
   530→        const W = 16 * (w + hh), H = 8 * (w + hh) + 2;
   531→        for (let yy = 0; yy < H; yy += 2) {
   532→          for (let xx = ((yy >> 1) % 2) * 2; xx < W * 2; xx += 4) {
   533→            const px = x0 - 16 * hh + xx, py = y0 + yy;
   534→            if (this.inFootprint(px - x0, py - y0, w, hh)) ctx.fillRect(px, py, 2, 1);
   535→          }
   536→        }
   537→      }
   538→    }
   539→  }

   543→  private renderMini(): void {
   544→    const mini = this.mini;
   545→    if (!mini) return;
   546→    const mctx = mini.getContext('2d')!;
   547→    const n = this.map.n;
   548→    if (this.miniDirty) {
   549→      const W = n * MM;
   550→      if (this.miniBase.width !== W) { this.miniBase.width = W; this.miniBase.height = W; }
   551→      const bctx = this.miniBase.getContext('2d')!;
   552→      const img = bctx.createImageData(W, W);
   553→      const u32 = new Uint32Array(img.data.buffer);
   554→      const abgr = (hex: string): number => {
   555→        const q = parseInt(hex.slice(1), 16);
   556→        return (255 << 24) | ((q & 255) << 16) | (((q >> 8) & 255) << 8) | ((q >> 16) & 255);
   557→      };
   558→      const C = {
   559→        grass: abgr('#3B7138'), water: abgr('#3467AC'), sand: abgr('#CFB379'),
   560→        farm: abgr('#B9C46B'), rock: abgr('#6E737F'), tree: abgr('#245129'),
   561→        sakura: abgr('#C97B93'), road: abgr('#4A4C55'),
   562→        snow: abgr('#DCE6F0'), flower: abgr('#A5C94F'), dry: abgr('#B99B62'),
   563→        swamp: abgr('#3F5F48'), autumn: abgr('#C9924E'),
   564→        res: abgr('#C25B44'), com: abgr('#E8CC4C'), ind: abgr('#9BA0A7'), civ: abgr('#5BC8B8'),
   565→      };
   566→      for (let y = 0; y < n; y++) {
   567→        for (let x = 0; x < n; x++) {
   568→          const i = this.map.idx(x, y);
   569→          const k = this.map.terr[i];
   570→          let cv: number = k === 1 ? C.water : k === 2 ? C.sand
   571→            : k === 3 ? C.farm : k === 4 ? C.rock
   572→            : k === 5 ? C.snow : k === 6 ? C.flower : k === 7 ? C.dry
   573→            : k === 8 ? C.swamp : k === 9 ? C.autumn : C.grass;
   574→          if (this.map.tree[i] >= 0) cv = this.map.tree[i] === 3 ? C.sakura : C.tree;
   575→          if (this.map.road[i]) cv = C.road;
   576→          const a = this.map.occ[i];
   577→          if (a >= 0) {
   578→            const o = this.map.objs.get(a)!;
   579→            const cat = defById(o.id).cat;
   580→            cv = cat === 'res' ? C.res : cat === 'com' ? C.com
   581→              : cat === 'ind' ? C.ind : C.civ;
   582→          }
   583→          for (let dy = 0; dy < MM; dy++) {
   584→            const row = (y * MM + dy) * W + x * MM;
   585→            for (let dx = 0; dx < MM; dx++) u32[row + dx] = cv;
   586→          }
   587→        }
   588→      }
   589→      bctx.putImageData(img, 0, 0);
   590→      this.miniDirty = false;
   591→    }
   592→    mctx.imageSmoothingEnabled = false;
   593→    mctx.drawImage(this.miniBase, 0, 0);
   594→    const vr = this.visibleRange();
   595→    mctx.strokeStyle = '#F4F2E8';
   596→    mctx.lineWidth = 1;
   597→    mctx.strokeRect(vr.x0 * MM + 0.5, vr.y0 * MM + 0.5, (vr.x1 - vr.x0) * MM, (vr.y1 - vr.y0) * MM);
   598→  }

   600→  private bindMini(mini: HTMLCanvasElement): void {
   601→    const jump = (e: PointerEvent): void => {
   602→      const rect = mini.getBoundingClientRect();
   603→      const tx = Math.floor(((e.clientX - rect.left) / rect.width) * this.map.n);
   604→      const ty = Math.floor(((e.clientY - rect.top) / rect.height) * this.map.n);
   605→      this.cam.x = (tx - ty) * 16;
   606→      this.cam.y = (tx + ty) * 8;
   607→    };
   608→    this.on(mini, 'pointerdown', (e) => {
   609→      jump(e as PointerEvent);
   610→      const mv = (ev: Event) => jump(ev as PointerEvent);
   611→      const up = () => {
   612→        mini.removeEventListener('pointermove', mv);
   613→        mini.removeEventListener('pointerup', up);
   614→      };
   615→      mini.addEventListener('pointermove', mv);
   616→      mini.addEventListener('pointerup', up);
   617→    });
   618→  }

   636→  private pinchSpan(): number {
   637→    const p = [...this.pointers.values()];
   638→    return p.length >= 2 ? Math.hypot(p[0].x - p[1].x, p[0].y - p[1].y) : 0;
   639→  }

   641→  private pinchMid(): { x: number; y: number } {
   642→    const p = [...this.pointers.values()];
   643→    return { x: (p[0].x + p[1].x) / 2, y: (p[0].y + p[1].y) / 2 };
   644→  }

   646→  private bindInput(display: HTMLCanvasElement): void {
   647→    this.on(display, 'contextmenu', (e) => e.preventDefault());
   648→    this.on(display, 'pointerdown', (e) => {
   649→      const pe = e as PointerEvent;
   650→      try { display.setPointerCapture(pe.pointerId); } catch { /* 无活动指针（合成事件等） */ }
   651→      this.pointers.set(pe.pointerId, { x: pe.clientX, y: pe.clientY });
   652→      // 双指手势：取消涂画，进入捏合缩放 / 平移
   653→      if (this.pointers.size >= 2) {
   654→        this.painting = false;
   655→        this.panning = false;
   656→        this.lastTile = null;
   657→        this.pinchDist = this.pinchSpan();
   658→        this.lastPinch = null;
   659→        return;
   660→      }
   661→      const rect = display.getBoundingClientRect();
   662→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   663→      this.beeper.play('click');
   664→      if (pe.button === 1 || pe.button === 2 || this.tool.t === 'view') {
   665→        this.panning = true;
   666→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   667→        return;
   668→      }
   669→      if (pe.button !== 0) return;
   670→      const { tx, ty } = this.tileAt(cx, cy);
   671→      this.pushUndo();
   672→      this.painting = true;
   673→      this.lastTile = { x: tx, y: ty };
   674→      this.applyToolAt(tx, ty);
   675→    });
   676→    this.on(display, 'pointermove', (e) => {
   677→      const pe = e as PointerEvent;
   678→      if (this.pointers.has(pe.pointerId)) {
   679→        this.pointers.set(pe.pointerId, { x: pe.clientX, y: pe.clientY });
   680→      }
   681→      // ---- 双指：中点平移 + 距离捏合（分档缩放）----
   682→      if (this.pointers.size >= 2) {
   683→        const mid = this.pinchMid();
   684→        if (this.lastPinch) {
   685→          this.cam.x -= (mid.x - this.lastPinch.x) / this.zoom;
   686→          this.cam.y -= (mid.y - this.lastPinch.y) / this.zoom;
   687→        }
   688→        this.lastPinch = mid;
   689→        const span = this.pinchSpan();
   690→        if (this.pinchDist > 24) {
   691→          const ratio = span / this.pinchDist;
   692→          const i = ZOOMS.indexOf(this.zoom);
   693→          if (ratio > 1.30 && i < ZOOMS.length - 1) {
   694→            const rect = display.getBoundingClientRect();
   695→            this.setZoom(ZOOMS[i + 1], { x: mid.x - rect.left, y: mid.y - rect.top });
   696→            this.pinchDist = span;
   697→          } else if (ratio < 0.77 && i > 0) {
   698→            const rect = display.getBoundingClientRect();
   699→            this.setZoom(ZOOMS[i - 1], { x: mid.x - rect.left, y: mid.y - rect.top });
   700→            this.pinchDist = span;
   701→          }
   702→        }
   703→        return;
   704→      }
   705→      const rect = display.getBoundingClientRect();
   706→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   707→      const { tx, ty } = this.tileAt(cx, cy);
   708→      this.hover = { tx, ty };
   709→      this.hooks.onHover?.(tx, ty);
   710→      if (this.panning) {
   711→        this.cam.x -= (pe.clientX - this.lastPan.x) / this.zoom;
   712→        this.cam.y -= (pe.clientY - this.lastPan.y) / this.zoom;
   713→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   714→        return;
   715→      }
   716→      if (this.painting) {
   717→        const lt = this.lastTile;
   718→        if (!lt) {
   719→          this.lastTile = { x: tx, y: ty };
   720→          this.dragMode = true;
   721→          this.applyToolAt(tx, ty);
   722→          this.dragMode = false;
   723→          return;
   724→        }
   725→        const steps = Math.max(Math.abs(tx - lt.x), Math.abs(ty - lt.y));
   726→        if (steps > 0) {
   727→          this.dragMode = true;
   728→          for (let s = 1; s <= steps; s++) {
   729→            const ix = Math.round(lt.x + (tx - lt.x) * s / steps);
   730→            const iy = Math.round(lt.y + (ty - lt.y) * s / steps);
   731→            this.applyToolAt(ix, iy);
   732→          }
   733→          this.dragMode = false;
   734→          this.lastTile = { x: tx, y: ty };
   735→        }
   736→      }
   737→    });
   738→    const endPointer = (e: Event): void => {
   739→      const pe = e as PointerEvent;
   740→      this.pointers.delete(pe.pointerId);
   741→      if (this.pointers.size < 2) {
   742→        this.pinchDist = 0;
   743→        this.lastPinch = null;
   744→      }
   745→      if (this.pointers.size === 0) {
   746→        this.painting = false;
   747→        this.panning = false;
   748→        this.lastTile = null;
   749→      }
   750→    };
   751→    this.on(display, 'pointerup', endPointer);
   752→    this.on(display, 'pointercancel', endPointer);
   753→    this.on(display, 'pointerleave', () => { this.hover = null; });
   754→    this.on(display, 'wheel', (e) => {
   755→      const we = e as WheelEvent;
   756→      we.preventDefault();
   757→      const rect = display.getBoundingClientRect();
   758→      this.setZoom(
   759→        ZOOMS[Math.max(0, Math.min(ZOOMS.length - 1, ZOOMS.indexOf(this.zoom) + (we.deltaY < 0 ? 1 : -1)))],
   760→        { x: we.clientX - rect.left, y: we.clientY - rect.top },
   761→      );
   762→    }, { passive: false });
   763→
   764→    this.on(window, 'keydown', (e) => {
   765→      const ke = e as KeyboardEvent;
   766→      if (ke.target instanceof HTMLInputElement || ke.target instanceof HTMLTextAreaElement) return;
   767→      this.keys.add(ke.code);
   768→      if (ke.code === 'Space') { e.preventDefault(); this.center(); }
   769→      if (ke.code === 'Equal' || ke.code === 'NumpadAdd') this.cycleZoom(1);
   770→      if (ke.code === 'Minus' || ke.code === 'NumpadSubtract') this.cycleZoom(-1);
   771→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyZ') {
   772→        e.preventDefault();
   773→        if (ke.shiftKey) this.redo(); else this.undo();
   774→      }
   775→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyY') {
   776→        e.preventDefault();
   777→        this.redo();
   778→      }
   779→      if (ke.code === 'Escape') this.setTool({ t: 'view' });
   780→    });
   781→    this.on(window, 'keyup', (e) => this.keys.delete((e as KeyboardEvent).code));
   782→  }

   786→  private applyToolAt(tx: number, ty: number): void {
   787→    const t = this.tool;
   788→    switch (t.t) {
   789→      case 'terr':
   790→        if (this.map.setTerr(tx, ty, t.k)) this.beeper.play('paint');
   791→        break;
   792→      case 'road':
   793→        if (this.map.setRoad(tx, ty, true)) this.beeper.play('paint');
   794→        break;
   795→      case 'tree':
   796→        if (this.map.plantTree(tx, ty)) this.beeper.play('paint');
   797→        break;
   798→      case 'doze':
   799→        if (this.map.bulldoze(tx, ty)) this.beeper.play('delete');
   800→        break;
   801→      case 'build': {
   802→        if (this.map.place(t.id, t.variant, tx, ty)) {
   803→          this.beeper.play('place');
   804→        } else if (!this.dragMode && this.map.inB(tx, ty)) {
   805→          this.beeper.play('error');
   806→        }
   807→        break;
   808→      }
   809→    }
   810→  }
