   130→  private on(t: EventTarget, name: string, f: EventListener, opts?: AddEventListenerOptions): void {

   135→  resize(): void {
   136→    const d = this.display;
   137→    if (!d) return;
   138→    const rect = d.getBoundingClientRect();
   139→    this.dpr = Math.min(2, window.devicePixelRatio || 1);
   140→    const cssW = Math.max(100, rect.width);
   141→    const cssH = Math.max(100, rect.height);
   142→    d.width = Math.round(cssW * this.dpr);
   143→    d.height = Math.round(cssH * this.dpr);
   144→    this.lowres.width = Math.ceil(cssW / this.zoom);
   145→    this.lowres.height = Math.ceil(cssH / this.zoom);
   146→    this.lctx.imageSmoothingEnabled = false;
   147→    if (this.dctx) this.dctx.imageSmoothingEnabled = false;

   150→  // ---------------- 坐标 ----------------

   152→  private get lw(): number { return this.lowres.width; }

   155→  /** 世界格角点 → 等距屏幕 px（低分辨率画布坐标） */
   156→  private w2sx(wx: number, wy: number): number { return (wx - wy) * 16 - this.cam.x + this.lw / 2; }
   157→  private w2sy(wx: number, wy: number): number { return (wx + wy) * 8 - this.cam.y + this.lh / 2; }
   158→
   159→  /** 鼠标 CSS 坐标 → 格坐标 */
   160→  tileAt(cssX: number, cssY: number): { tx: number; ty: number } {
   161→    const sx = cssX / this.zoom, sy = cssY / this.zoom;
   162→    const ax = sx + this.cam.x - this.lw / 2;   // = (X−Y)*16
   163→    const ay = sy + this.cam.y - this.lh / 2;   // = (X+Y)*8
   164→    const X = ax / 32 + ay / 16;
   165→    const Y = ay / 16 - ax / 32;
   166→    return { tx: Math.floor(X), ty: Math.floor(Y) };
   167→  }
   168→
   169→  center(): void {
   170→    const c = this.map.n / 2;
   171→    this.cam.x = (c - c) * 16;
   172→    this.cam.y = (c + c) * 8;
   173→  }

   194→  cycleZoom(dir: number): void {
   195→    const i = ZOOMS.indexOf(this.zoom);
   196→    const j = Math.max(0, Math.min(ZOOMS.length - 1, i + dir));
   197→    if (j !== i) this.setZoom(ZOOMS[j]);
   198→  }
   199→
   200→  setZoom(z: number, focusCss?: { x: number; y: number }): void {
   201→    if (z === this.zoom) return;
   202→    let keep: { gx: number; gy: number } | null = null;
   203→    if (focusCss) {
   204→      const t = this.tileAt(focusCss.x, focusCss.y);
   205→      keep = { gx: t.tx + 0.5, gy: t.ty + 0.5 };
   206→    }
   207→    this.zoom = z;
   208→    this.resize();
   209→    if (keep && focusCss) {
   210→      const sx = focusCss.x / z, sy = focusCss.y / z;
   211→      this.cam.x = (keep.gx - keep.gy) * 16 - sx + this.lw / 2;
   212→      this.cam.y = (keep.gx + keep.gy) * 8 - sy + this.lh / 2;
   213→    }
   214→    this.beeper.play('zoom');
   215→    this.hooks.onZoom?.(z);
   216→  }
   217→
   218→  // ---------------- 主循环 ----------------
   219→
   220→  private loop = (): void => {
   221→    this.raf = requestAnimationFrame(this.loop);
   222→    const now = performance.now();
   223→    const dt = Math.min(100, now - this.lastT);
   224→    this.lastT = now;
   225→    this.fpsAcc += dt; this.fpsN++;
   226→    if (now - this.fpsAt > 500) {
   227→      this.hooks.onFps?.(Math.round(1000 / (this.fpsAcc / this.fpsN)));
   228→      this.fpsAcc = 0; this.fpsN = 0; this.fpsAt = now;
   229→    }
   230→    const pan = dt * 0.72;
   231→    if (this.keys.has('ArrowLeft') || this.keys.has('KeyA')) this.cam.x -= pan;
   232→    if (this.keys.has('ArrowRight') || this.keys.has('KeyD')) this.cam.x += pan;
   233→    if (this.keys.has('ArrowUp') || this.keys.has('KeyW')) this.cam.y -= pan;
   234→    if (this.keys.has('ArrowDown') || this.keys.has('KeyS')) this.cam.y += pan;
   235→    if (this.crowds) this.actors.tick(dt / 1000, this.map);
   236→    this.render(now);
   237→    if (this.map.version !== this.lastSaveVer && now - this.lastSaveT > 25000) {
   238→      this.autoSave();
   239→    }
   240→  };

   244→  private visibleRange(): { x0: number; y0: number; x1: number; y1: number } {
   245→    const n = this.map.n;
   246→    const cs: Array<[number, number]> = [[0, 0], [this.lw, 0], [0, this.lh], [this.lw, this.lh]];
   247→    let minX = n, maxX = 0, minY = n, maxY = 0;
   248→    for (const [sx, sy] of cs) {
   249→      const ax = sx + this.cam.x - this.lw / 2;
   250→      const ay = sy + this.cam.y - this.lh / 2;
   251→      const X = ax / 32 + ay / 16;
   252→      const Y = ay / 16 - ax / 32;
   253→      minX = Math.min(minX, Math.floor(X)); maxX = Math.max(maxX, Math.ceil(X));
   254→      minY = Math.min(minY, Math.floor(Y)); maxY = Math.max(maxY, Math.ceil(Y));
   255→    }
   256→    const pad = 4;
   257→    return {
   258→      x0: Math.max(0, minX - pad), y0: Math.max(0, minY - pad),
   259→      x1: Math.min(n - 1, maxX + pad), y1: Math.min(n - 1, maxY + pad),
   260→    };
   261→  }

   263→  private render(t: number): void {
   264→    const ctx = this.lctx;
   265→    const { lw, lh } = this;
   266→
   267→    // 背景虚空（深海蓝 + 稀疏星点抖动）
   268→    ctx.fillStyle = '#0B0F24';
   269→    ctx.fillRect(0, 0, lw, lh);
   270→    ctx.fillStyle = '#182142';
   271→    for (let y = 0; y < lh; y += 8) {
   272→      for (let x = ((y >> 3) % 2) * 8; x < lw; x += 16) ctx.fillRect(x + 3, y + 5, 1, 1);
   273→    }
   274→
   275→    const vr = this.visibleRange();
   276→    const wf = Math.floor(t / 620) % 2;
   277→    const items: RenderItem[] = [];
   278→    const m = this.map;
   279→
   280→    // ---- Pass 1：地面 ----
   281→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   282→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   283→        const i = m.idx(tx, ty);
   284→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   285→        if (sx < -96 || sx > lw + 96 || sy < -160 || sy > lh + 96) continue;
   286→
   287→        const a = m.occ[i];
   288→        let groundDrawn = false;
   289→        if (a >= 0) {
   290→          const o = m.objs.get(a)!;
   291→          const d = defById(o.id);
   292→          const isAnchor = o.ax === tx && o.ay === ty;
   293→          if (d.ground) {
   294→            if (isAnchor) {
   295→              const g = getGround(d.ground, { variant: o.v });
   296→              ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   297→            }
   298→            groundDrawn = true;   // 大地面精灵覆盖全部占地格
   299→          }
   300→        }
   301→        if (groundDrawn) continue;
   302→
   303→        if (m.road[i]) {
   304→          const g = getGround('road', { road: this.roadMask(tx, ty), variant: Math.floor(hash2(tx, ty, 5) * 4) });
   305→          ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   306→        } else {
   307→          const k = m.terr[i];
   308→          const v = Math.floor(hash2(tx, ty, 9) * 4);
   309→          if (k === TERR_WATER) {
   310→            const g = getGround('water', { edge: this.landEdge(tx, ty), variant: v, frame: wf });
   311→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   312→          } else if (k === TERR_SAND) {
   313→            const g = getGround('sand', { variant: v });
   314→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   315→          } else if (k === TERR_FARM) {
   316→            const g = getGround('farm', { variant: v, edge: this.waterEdge(tx, ty) });
   317→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   318→          } else if (k === TERR_ROCK) {
   319→            const g = getGround('rock', { variant: v, edge: this.waterEdge(tx, ty) });
   320→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   321→          } else {
   322→            const g = getGround('grass', { variant: v, edge: this.waterEdge(tx, ty) });
   323→            ctx.drawImage(g.cv, Math.round(sx + g.ox), Math.round(sy + g.oy));
   324→          }
   325→        }
   326→      }
   327→    }
   328→
   329→    // ---- Pass 2：树木 + 建筑（按前角深度排序）----
   330→    const seen = new Set<number>();
   331→    for (let ty = vr.y0; ty <= vr.y1; ty++) {
   332→      for (let tx = vr.x0; tx <= vr.x1; tx++) {
   333→        const i = m.idx(tx, ty);
   334→        const sx = this.w2sx(tx, ty), sy = this.w2sy(tx, ty);
   335→        if (sx < -160 || sx > lw + 160 || sy < -300 || sy > lh + 160) continue;
   336→
   337→        const tv = m.tree[i];
   338→        if (tv >= 0 && !m.road[i] && m.occ[i] < 0) {
   339→          const sp = this.treeSprites[tv % this.treeSprites.length];
   340→          const key = tx + ty;
   341→          const dx = Math.round(sx + sp.ox), dy = Math.round(sy + sp.oy);
   342→          items.push({ key, draw: () => ctx.drawImage(sp.cv, dx, dy) });
   343→        }
   344→
   345→        const a = m.occ[i];
   346→        if (a >= 0 && !seen.has(a)) {
   347→          seen.add(a);
   348→          const o = m.objs.get(a)!;
   349→          const d = defById(o.id);
   350→          const bs = this.book.get(o.id, o.v);
   351→          const sp = bs.sp;
   352→          const bsx = this.w2sx(o.ax, o.ay), bsy = this.w2sy(o.ax, o.ay);
   353→          const key = (o.ax + d.w - 1) + (o.ay + d.h - 1);
   354→          const blinkOn = Math.floor(t / 640) % 2 === 0;
   355→          const smokeFr = Math.floor(t / 300) % 3;
   356→          const bx = Math.round(bsx + sp.ox), by = Math.round(bsy + sp.oy);
   357→          const smk = bs.smokeAt;
   358→          const blk = bs.blinkAt;
   359→          const sm = smk ? this.book.smoke[smokeFr] : null;
   360→          const mx = smk ? bsx + (smk[0] - smk[1]) : 0;
   361→          const my = smk ? bsy + (smk[0] + smk[1]) / 2 - smk[2] : 0;
   362→          const lx = blk ? bsx + (blk[0] - blk[1]) : 0;
   363→          const ly = blk ? bsy + (blk[0] + blk[1]) / 2 - blk[2] : 0;
   364→          items.push({
   365→            key,
   366→            draw: () => {
   367→              ctx.drawImage(sp.cv, bx, by);
   368→              if (sm) ctx.drawImage(sm.cv, Math.round(mx + sm.ox), Math.round(my + sm.oy));
   369→              if (blk && blinkOn) {
   370→                ctx.fillStyle = PALETTE[62];
   371→                ctx.fillRect(Math.round(lx) - 1, Math.round(ly) - 1, 2, 2);
   372→                ctx.fillStyle = PALETTE[61];
   373→                ctx.fillRect(Math.round(lx), Math.round(ly), 1, 1);
   374→              }
   375→            },
   376→          });
   377→        }
   378→      }
   379→    }
   380→    // ---- 行人 / 车辆（与建筑同序参与深度排序）----
   381→    if (this.crowds) {
   382→      for (const a of this.actors.actors) {
   383→        const px = a.fx + a.dx * a.t, py = a.fy + a.dy * a.t;
   384→        let wx: number, wy: number;
   385→        if (a.kind === 0) {
   386→          wx = (px + 0.5) * 16 + a.offX;
   387→          wy = (py + 0.5) * 16 + a.offY;
   388→        } else {
   389→          wx = px * 16 - a.dy * 2.3;
   390→          wy = py * 16 + a.dx * 2.3;
   391→        }
   392→        const sx = (wx - wy) - this.cam.x + lw / 2;
   393→        const sy = (wx + wy) / 2 - this.cam.y + lh / 2;
   394→        if (sx < -80 || sx > lw + 80 || sy < -60 || sy > lh + 60) continue;
   395→        const key = (wx + wy) / 16;
   396→        if (a.kind === 0) {
   397→          const d = ActorSystem.dirOf(a);
   398→          const fr = Math.floor(t / 220 + a.phase) % 2;
   399→          const cv = this.actors.personCv(d as 'dr' | 'dl' | 'ur' | 'ul', fr, a.look);
   400→          if (!cv) continue;
   401→          const dx = Math.round(sx - 3), dy = Math.round(sy - 12);
   402→          items.push({ key, draw: () => ctx.drawImage(cv, dx, dy) });
   403→        } else {
   404→          const d = ActorSystem.dirOf(a);
   405→          const sp = this.actors.carSp(d as 'e' | 'w' | 's' | 'n', a.look);
   406→          if (!sp) continue;
   407→          const dx = Math.round(sx + sp.ox), dy = Math.round(sy + sp.oy);
   408→          items.push({ key, draw: () => ctx.drawImage(sp.cv, dx, dy) });
   409→        }
   410→      }
   411→    }
   412→
   413→    items.sort((p, q) => p.key - q.key);
   414→    for (const it of items) it.draw();
   415→
   416→    // ---- Pass 3：光标 + 幽灵 ----
   417→    this.drawCursor(ctx);
   418→
   419→    // ---- 上屏（最近邻放大 + 扫描线）----
   420→    const d = this.display;
   421→    const dctx = this.dctx;
   422→    if (d && dctx) {
   423→      const S = this.zoom * this.dpr;
   424→      dctx.imageSmoothingEnabled = false;
   425→      dctx.drawImage(this.lowres, 0, 0, lw, lh, 0, 0, Math.round(lw * S), Math.round(lh * S));
   426→      const step = Math.max(2, Math.round(S));
   427→      dctx.globalAlpha = 0.15;
   428→      dctx.fillStyle = '#000';
   429→      for (let y = step - 1; y < d.height; y += step) dctx.fillRect(0, y, d.width, 1);
   430→      dctx.globalAlpha = 1;
   431→    }
   432→
   433→    this.renderMini();
   434→  }
   435→
   436→  private roadMask(tx: number, ty: number): number {
   437→    const m = this.map;
   438→    let v = 0;
   439→    if (m.inB(tx, ty - 1) && m.road[m.idx(tx, ty - 1)]) v |= 1;
   440→    if (m.inB(tx + 1, ty) && m.road[m.idx(tx + 1, ty)]) v |= 2;
   441→    if (m.inB(tx, ty + 1) && m.road[m.idx(tx, ty + 1)]) v |= 4;
   442→    if (m.inB(tx - 1, ty) && m.road[m.idx(tx - 1, ty)]) v |= 8;
   443→    return v;
   444→  }
   445→
   446→  private waterEdge(tx: number, ty: number): number {
   447→    const m = this.map;
   448→    let v = 0;
   449→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] === TERR_WATER) v |= 1;
   450→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] === TERR_WATER) v |= 2;
   451→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] === TERR_WATER) v |= 4;
   452→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] === TERR_WATER) v |= 8;
   453→    return v;
   454→  }
   455→
   456→  private landEdge(tx: number, ty: number): number {
   457→    const m = this.map;
   458→    let v = 0;
   459→    if (!m.inB(tx, ty - 1) || m.terr[m.idx(tx, ty - 1)] !== TERR_WATER) v |= 1;
   460→    if (!m.inB(tx + 1, ty) || m.terr[m.idx(tx + 1, ty)] !== TERR_WATER) v |= 2;
   461→    if (!m.inB(tx, ty + 1) || m.terr[m.idx(tx, ty + 1)] !== TERR_WATER) v |= 4;
   462→    if (!m.inB(tx - 1, ty) || m.terr[m.idx(tx - 1, ty)] !== TERR_WATER) v |= 8;
   463→    return v;
   464→  }

   466→  /** 屏幕偏移是否在 w×h 足印菱形内（相对锚点） */
   467→  private inFootprint(dx: number, dy: number, w: number, h: number): boolean {
   468→    const X = dx / 2 + dy;
   469→    const Y = dy - dx / 2;
   470→    return X >= 0 && X <= 16 * w && Y >= 0 && Y <= 16 * h;
   471→  }

   473→  /** 光标菱形 + 放置幽灵 */
   474→  private drawCursor(ctx: CanvasRenderingContext2D): void {
   475→    const h = this.hover;
   476→    if (!h || !this.map.inB(h.tx, h.ty)) return;
   477→    const tool = this.tool;
   478→    let w = 1, hh = 1;
   479→    if (tool.t === 'build') {
   480→      const d = defById(tool.id);
   481→      w = d.w; hh = d.h;
   482→    }
   483→    const x0 = this.w2sx(h.tx, h.ty), y0 = this.w2sy(h.tx, h.ty);
   484→    ctx.beginPath();
   485→    ctx.moveTo(x0, y0);
   486→    ctx.lineTo(x0 + 16 * w, y0 + 8 * w);
   487→    ctx.lineTo(x0 + 16 * (w - hh), y0 + 8 * (w + hh));
   488→    ctx.lineTo(x0 - 16 * hh, y0 + 8 * hh);
   489→    ctx.closePath();
   490→    ctx.strokeStyle = 'rgba(8,10,18,0.85)';
   491→    ctx.lineWidth = 3;
   492→    ctx.stroke();
   493→    ctx.strokeStyle = '#F4F2E8';
   494→    ctx.lineWidth = 1;
   495→    ctx.stroke();
   496→
   497→    if (tool.t === 'build') {
   498→      const bs = this.book.get(tool.id, tool.variant);
   499→      const g = ghostSprite(bs.sp);
   500→      const ok = this.map.canPlace(tool.id, h.tx, h.ty);
   501→      ctx.drawImage(g, Math.round(x0 + bs.sp.ox), Math.round(y0 + bs.sp.oy));
   502→      if (!ok) {
   503→        ctx.fillStyle = 'rgba(224,85,72,0.55)';
   504→        const W = 16 * (w + hh), H = 8 * (w + hh) + 2;
   505→        for (let yy = 0; yy < H; yy += 2) {
   506→          for (let xx = ((yy >> 1) % 2) * 2; xx < W * 2; xx += 4) {
   507→            const px = x0 - 16 * hh + xx, py = y0 + yy;
   508→            if (this.inFootprint(px - x0, py - y0, w, hh)) ctx.fillRect(px, py, 2, 1);
   509→          }
   510→        }
   511→      }
   512→    }
   513→  }

   517→  private renderMini(): void {
   518→    const mini = this.mini;
   519→    if (!mini) return;
   520→    const mctx = mini.getContext('2d')!;
   521→    const n = this.map.n;
   522→    if (this.miniDirty) {
   523→      const W = n * MM;
   524→      if (this.miniBase.width !== W) { this.miniBase.width = W; this.miniBase.height = W; }
   525→      const bctx = this.miniBase.getContext('2d')!;
   526→      const img = bctx.createImageData(W, W);
   527→      const u32 = new Uint32Array(img.data.buffer);
   528→      const abgr = (hex: string): number => {
   529→        const q = parseInt(hex.slice(1), 16);
   530→        return (255 << 24) | ((q & 255) << 16) | (((q >> 8) & 255) << 8) | ((q >> 16) & 255);
   531→      };
   532→      const C = {
   533→        grass: abgr('#3B7138'), water: abgr('#3467AC'), sand: abgr('#CFB379'),
   534→        farm: abgr('#B9C46B'), rock: abgr('#6E737F'), tree: abgr('#245129'),
   535→        sakura: abgr('#C97B93'), road: abgr('#4A4C55'),
   536→        res: abgr('#C25B44'), com: abgr('#E8CC4C'), ind: abgr('#9BA0A7'), civ: abgr('#5BC8B8'),
   537→      };
   538→      for (let y = 0; y < n; y++) {
   539→        for (let x = 0; x < n; x++) {
   540→          const i = this.map.idx(x, y);
   541→          const k = this.map.terr[i];
   542→          let cv: number = k === 1 ? C.water : k === 2 ? C.sand
   543→            : k === 3 ? C.farm : k === 4 ? C.rock : C.grass;
   544→          if (this.map.tree[i] >= 0) cv = this.map.tree[i] === 3 ? C.sakura : C.tree;
   545→          if (this.map.road[i]) cv = C.road;
   546→          const a = this.map.occ[i];
   547→          if (a >= 0) {
   548→            const o = this.map.objs.get(a)!;
   549→            const cat = defById(o.id).cat;
   550→            cv = cat === 'res' ? C.res : cat === 'com' ? C.com
   551→              : cat === 'ind' ? C.ind : C.civ;
   552→          }
   553→          for (let dy = 0; dy < MM; dy++) {
   554→            const row = (y * MM + dy) * W + x * MM;
   555→            for (let dx = 0; dx < MM; dx++) u32[row + dx] = cv;
   556→          }
   557→        }
   558→      }
   559→      bctx.putImageData(img, 0, 0);
   560→      this.miniDirty = false;
   561→    }
   562→    mctx.imageSmoothingEnabled = false;
   563→    mctx.drawImage(this.miniBase, 0, 0);
   564→    const vr = this.visibleRange();
   565→    mctx.strokeStyle = '#F4F2E8';
   566→    mctx.lineWidth = 1;
   567→    mctx.strokeRect(vr.x0 * MM + 0.5, vr.y0 * MM + 0.5, (vr.x1 - vr.x0) * MM, (vr.y1 - vr.y0) * MM);
   568→  }

   570→  private bindMini(mini: HTMLCanvasElement): void {
   571→    const jump = (e: PointerEvent): void => {
   572→      const rect = mini.getBoundingClientRect();
   573→      const tx = Math.floor(((e.clientX - rect.left) / rect.width) * this.map.n);
   574→      const ty = Math.floor(((e.clientY - rect.top) / rect.height) * this.map.n);
   575→      this.cam.x = (tx - ty) * 16;
   576→      this.cam.y = (tx + ty) * 8;
   577→    };
   578→    this.on(mini, 'pointerdown', (e) => {
   579→      jump(e as PointerEvent);
   580→      const mv = (ev: Event) => jump(ev as PointerEvent);
   581→      const up = () => {
   582→        mini.removeEventListener('pointermove', mv);
   583→        mini.removeEventListener('pointerup', up);
   584→      };
   585→      mini.addEventListener('pointermove', mv);
   586→      mini.addEventListener('pointerup', up);
   587→    });
   588→  }

   606→  private pinchSpan(): number {
   607→    const p = [...this.pointers.values()];
   608→    return p.length >= 2 ? Math.hypot(p[0].x - p[1].x, p[0].y - p[1].y) : 0;
   609→  }

   611→  private pinchMid(): { x: number; y: number } {
   612→    const p = [...this.pointers.values()];
   613→    return { x: (p[0].x + p[1].x) / 2, y: (p[0].y + p[1].y) / 2 };
   614→  }

   616→  private bindInput(display: HTMLCanvasElement): void {
   617→    this.on(display, 'contextmenu', (e) => e.preventDefault());
   618→    this.on(display, 'pointerdown', (e) => {
   619→      const pe = e as PointerEvent;
   620→      try { display.setPointerCapture(pe.pointerId); } catch { /* 无活动指针（合成事件等） */ }
   621→      this.pointers.set(pe.pointerId, { x: pe.clientX, y: pe.clientY });
   622→      // 双指手势：取消涂画，进入捏合缩放 / 平移
   623→      if (this.pointers.size >= 2) {
   624→        this.painting = false;
   625→        this.panning = false;
   626→        this.lastTile = null;
   627→        this.pinchDist = this.pinchSpan();
   628→        this.lastPinch = null;
   629→        return;
   630→      }
   631→      const rect = display.getBoundingClientRect();
   632→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   633→      this.beeper.play('click');
   634→      if (pe.button === 1 || pe.button === 2 || this.tool.t === 'view') {
   635→        this.panning = true;
   636→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   637→        return;
   638→      }
   639→      if (pe.button !== 0) return;
   640→      const { tx, ty } = this.tileAt(cx, cy);
   641→      this.pushUndo();
   642→      this.painting = true;
   643→      this.lastTile = { x: tx, y: ty };
   644→      this.applyToolAt(tx, ty);
   645→    });
   646→    this.on(display, 'pointermove', (e) => {
   647→      const pe = e as PointerEvent;
   648→      if (this.pointers.has(pe.pointerId)) {
   649→        this.pointers.set(pe.pointerId, { x: pe.clientX, y: pe.clientY });
   650→      }
   651→      // ---- 双指：中点平移 + 距离捏合（分档缩放）----
   652→      if (this.pointers.size >= 2) {
   653→        const mid = this.pinchMid();
   654→        if (this.lastPinch) {
   655→          this.cam.x -= (mid.x - this.lastPinch.x) / this.zoom;
   656→          this.cam.y -= (mid.y - this.lastPinch.y) / this.zoom;
   657→        }
   658→        this.lastPinch = mid;
   659→        const span = this.pinchSpan();
   660→        if (this.pinchDist > 24) {
   661→          const ratio = span / this.pinchDist;
   662→          const i = ZOOMS.indexOf(this.zoom);
   663→          if (ratio > 1.30 && i < ZOOMS.length - 1) {
   664→            const rect = display.getBoundingClientRect();
   665→            this.setZoom(ZOOMS[i + 1], { x: mid.x - rect.left, y: mid.y - rect.top });
   666→            this.pinchDist = span;
   667→          } else if (ratio < 0.77 && i > 0) {
   668→            const rect = display.getBoundingClientRect();
   669→            this.setZoom(ZOOMS[i - 1], { x: mid.x - rect.left, y: mid.y - rect.top });
   670→            this.pinchDist = span;
   671→          }
   672→        }
   673→        return;
   674→      }
   675→      const rect = display.getBoundingClientRect();
   676→      const cx = pe.clientX - rect.left, cy = pe.clientY - rect.top;
   677→      const { tx, ty } = this.tileAt(cx, cy);
   678→      this.hover = { tx, ty };
   679→      this.hooks.onHover?.(tx, ty);
   680→      if (this.panning) {
   681→        this.cam.x -= (pe.clientX - this.lastPan.x) / this.zoom;
   682→        this.cam.y -= (pe.clientY - this.lastPan.y) / this.zoom;
   683→        this.lastPan = { x: pe.clientX, y: pe.clientY };
   684→        return;
   685→      }
   686→      if (this.painting) {
   687→        const lt = this.lastTile;
   688→        if (!lt) {
   689→          this.lastTile = { x: tx, y: ty };
   690→          this.dragMode = true;
   691→          this.applyToolAt(tx, ty);
   692→          this.dragMode = false;
   693→          return;
   694→        }
   695→        const steps = Math.max(Math.abs(tx - lt.x), Math.abs(ty - lt.y));
   696→        if (steps > 0) {
   697→          this.dragMode = true;
   698→          for (let s = 1; s <= steps; s++) {
   699→            const ix = Math.round(lt.x + (tx - lt.x) * s / steps);
   700→            const iy = Math.round(lt.y + (ty - lt.y) * s / steps);
   701→            this.applyToolAt(ix, iy);
   702→          }
   703→          this.dragMode = false;
   704→          this.lastTile = { x: tx, y: ty };
   705→        }
   706→      }
   707→    });
   708→    const endPointer = (e: Event): void => {
   709→      const pe = e as PointerEvent;
   710→      this.pointers.delete(pe.pointerId);
   711→      if (this.pointers.size < 2) {
   712→        this.pinchDist = 0;
   713→        this.lastPinch = null;
   714→      }
   715→      if (this.pointers.size === 0) {
   716→        this.painting = false;
   717→        this.panning = false;
   718→        this.lastTile = null;
   719→      }
   720→    };
   721→    this.on(display, 'pointerup', endPointer);
   722→    this.on(display, 'pointercancel', endPointer);
   723→    this.on(display, 'pointerleave', () => { this.hover = null; });
   724→    this.on(display, 'wheel', (e) => {
   725→      const we = e as WheelEvent;
   726→      we.preventDefault();
   727→      const rect = display.getBoundingClientRect();
   728→      this.setZoom(
   729→        ZOOMS[Math.max(0, Math.min(ZOOMS.length - 1, ZOOMS.indexOf(this.zoom) + (we.deltaY < 0 ? 1 : -1)))],
   730→        { x: we.clientX - rect.left, y: we.clientY - rect.top },
   731→      );
   732→    }, { passive: false });
   733→
   734→    this.on(window, 'keydown', (e) => {
   735→      const ke = e as KeyboardEvent;
   736→      if (ke.target instanceof HTMLInputElement || ke.target instanceof HTMLTextAreaElement) return;
   737→      this.keys.add(ke.code);
   738→      if (ke.code === 'Space') { e.preventDefault(); this.center(); }
   739→      if (ke.code === 'Equal' || ke.code === 'NumpadAdd') this.cycleZoom(1);
   740→      if (ke.code === 'Minus' || ke.code === 'NumpadSubtract') this.cycleZoom(-1);
   741→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyZ') {
   742→        e.preventDefault();
   743→        if (ke.shiftKey) this.redo(); else this.undo();
   744→      }
   745→      if ((ke.ctrlKey || ke.metaKey) && ke.code === 'KeyY') {
   746→        e.preventDefault();
   747→        this.redo();
   748→      }
   749→      if (ke.code === 'Escape') this.setTool({ t: 'view' });
   750→    });
   751→    this.on(window, 'keyup', (e) => this.keys.delete((e as KeyboardEvent).code));
   752→  }

   756→  private applyToolAt(tx: number, ty: number): void {
   757→    const t = this.tool;
   758→    switch (t.t) {
   759→      case 'terr':
   760→        if (this.map.setTerr(tx, ty, t.k)) this.beeper.play('paint');
   761→        break;
   762→      case 'road':
   763→        if (this.map.setRoad(tx, ty, true)) this.beeper.play('paint');
   764→        break;
   765→      case 'tree':
   766→        if (this.map.plantTree(tx, ty)) this.beeper.play('paint');
   767→        break;
   768→      case 'doze':
   769→        if (this.map.bulldoze(tx, ty)) this.beeper.play('delete');
   770→        break;
   771→      case 'build': {
   772→        if (this.map.place(t.id, t.variant, tx, ty)) {
   773→          this.beeper.play('place');
   774→        } else if (!this.dragMode && this.map.inB(tx, ty)) {
   775→          this.beeper.play('error');
   776→        }
   777→        break;
   778→      }
   779→    }
   780→  }
