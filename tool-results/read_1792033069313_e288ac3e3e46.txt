    99→const UI_KEY = 'pxc_ui_v1';
   100→
   101→function loadUi(): UiState {
   102→  try {
   103→    const raw = localStorage.getItem(UI_KEY);
   104→    if (raw) return { ...UI_DEFAULT, ...(JSON.parse(raw) as Partial<UiState>) };
   105→  } catch { /* 忽略 */ }
   106→  return { ...UI_DEFAULT };
   107→}
   108→
   109→// ---- 外部 UI 状态存储（localStorage 持久化，useSyncExternalStore 水合安全）----
   110→let uiStore: UiState = (() => {
   111→  try {
   112→    const raw = localStorage.getItem(UI_KEY);
   113→    if (raw) return { ...UI_DEFAULT, ...(JSON.parse(raw) as Partial<UiState>) };
   114→  } catch { /* SSR / 数据损坏时用默认值 */ }
   115→  return { ...UI_DEFAULT };
   116→})();
   117→const uiSubs = new Set<() => void>();
   118→
   119→function setUiStore(next: (u: UiState) => UiState): void {
   120→  uiStore = next(uiStore);
   121→  try { localStorage.setItem(UI_KEY, JSON.stringify(uiStore)); } catch { /* 忽略 */ }
   122→  for (const fn of uiSubs) fn();
   123→}
   124→
   125→function useUiState(): [UiState, typeof setUiStore] {
   126→  const subscribe = useCallback((cb: () => void) => {
   127→    uiSubs.add(cb);
   128→    return () => { uiSubs.delete(cb); };
   129→  }, []);
   130→  const snap = useSyncExternalStore(subscribe, () => uiStore, () => UI_DEFAULT);
   131→  return [snap, setUiStore];
   132→}

   134→function toolHint(t: Tool): string {
   135→  switch (t.t) {
   136→    case 'view': return '拖动平移 · 滚轮/双指缩放 · 空格居中';
   137→    case 'terr': return `按住拖动画画「${TERR_OPTS[t.k]?.label ?? '地形'}」（水域会淹没地上物）`;
   138→    case 'tree': return '点击或拖动，种下随机小树（共 6 种，含樱花与棕榈）';
   139→    case 'road': return '按住拖出路网，转角自动拼接';
   140→    case 'doze': return '点击拆除：整栋建筑 / 树木 / 道路';
   141→    case 'build': {
   142→      const d = defById(t.id);
   143→      return `点击放置「${d.name}」占地 ${d.w}×${d.h}${d.pop ? ` · 容纳 ${d.pop} 人` : ''}`;
   144→    }
   145→  }
